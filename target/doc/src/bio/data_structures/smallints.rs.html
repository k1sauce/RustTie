6//! A data structure for a sequence of small integers with a few big integers.
7//! Small ints are stored in type S (e.g. a byte), big ints are stored separately (in type `B`) in a BTree.
8//! The implementation provides vector-like operations on the data structure (e.g. retrieve a position,
9//! add an integer, etc.). Getting and setting (by position) time complexity is `O(1)` for small ints,
10//! and `O(b)` for big ints, where `b` is the number of big ints stored.
11//!
12//! # Space usage
13//! SmallInts pay the cost of slower retrieval of big integers with smaller overall memory usage;
14//! `O(size_of(S) * (s+b) + size_of(B) * b)` where `S` and `B` are the small and large int types, and
15//! `s` and `b` are the number of those stored respectively.
16//!
17//! # Example
18//!
19//! ```
20//! use bio::data_structures::smallints::SmallInts;
21//! let mut smallints: SmallInts<u8, usize> = SmallInts::new();
22//! smallints.push(3);
23//! smallints.push(4);
24//! smallints.push(255);
25//! smallints.push(305093);
26//! assert_eq!(smallints.get(0).unwrap(), 3);
27//! smallints.set(0, 50000);
28//! let values: Vec<usize> = smallints.iter().collect();
29//! assert_eq!(values, [50000, 4, 255, 305093]);
30//! ```
31
32use std::collections::BTreeMap;
33use std::iter::{repeat, Enumerate};
34use std::mem::size_of;
35use std::slice;
36
37use num_integer::Integer;
38use num_traits::{cast, Bounded, Num, NumCast};

40/// Data structure for storing a sequence of small integers with few big ones space efficiently
41/// while supporting classical vector operations.
42#[derive(Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize)]
43pub struct SmallInts<F: Integer + Bounded + NumCast + Copy, B: Integer + NumCast + Copy> {
44    smallints: Vec<F>,
45    bigints: BTreeMap<usize, B>,
46}
47
48impl<S: Integer + Bounded + NumCast + Copy, B: Integer + NumCast + Copy> Default
49    for SmallInts<S, B>
50{
51    fn default() -> Self {
52        assert!(
53            size_of::<S>() < size_of::<B>(),
54            "S has to be smaller than B"
55        );
56        SmallInts {
57            smallints: Vec::new(),
58            bigints: BTreeMap::new(),
59        }
60    }
61}
62
63impl<S: Integer + Bounded + NumCast + Copy, B: Integer + NumCast + Copy> SmallInts<S, B> {
64    /// Create a new instance.
65    pub fn new() -> Self {
66        Default::default()
67    }
68
69    /// Create a new instance with a given capacity.
70    pub fn with_capacity(n: usize) -> Self {
71        assert!(
72            size_of::<S>() < size_of::<B>(),
73            "S has to be smaller than B"
74        );
75        SmallInts {
76            smallints: Vec::with_capacity(n),
77            bigints: BTreeMap::new(),
78        }
79    }
80
81    /// Create a new instance containing `n` times the integer `v` (and `v` is expected to be small).
82    pub fn from_elem(v: S, n: usize) -> Self {
83        assert!(
84            size_of::<S>() < size_of::<B>(),
85            "S has to be smaller than B"
86        );
87        if v > cast(0).unwrap() {
88            assert!(v < S::max_value(), "v has to be smaller than maximum value");
89        }
90
91        SmallInts {
92            smallints: repeat(v).take(n).collect(),
93            bigints: BTreeMap::new(),
94        }
95    }
96
97    /// Return the integer at position `i`. Time complexity `O(1)` if `i` points to a small int,
98    /// `O(log(b))` for a big int, where `b` denotes the number of big ints stored.
99    pub fn get(&self, i: usize) -> Option<B> {
100        if i < self.smallints.len() {
101            self.real_value(i, self.smallints[i])
102        } else {
103            None
104        }
105    }
106
107    /// Append `v` to the sequence. This will determine whether `v` is big or small and store it
108    /// accordingly. Time complexity `O(1)` for small ints and `O(log(b))` for big ints,
109    /// where `b` denotes the number of big ints stored.
110    pub fn push(&mut self, v: B) {
111        let maxv: S = S::max_value();
112        match cast(v) {
113            Some(v) if v < maxv => self.smallints.push(v),
114            _ => {
115                let i = self.smallints.len();
116                self.smallints.push(maxv);
117                self.bigints.insert(i, v);
118            }
119        }
120    }
121
122    /// Set value of position `i` to `v`. This will determine whether `v` is big or small and store it accordingly.
123    /// Time complexity `O(1)` for small ints and `O(log(b))` for big ints,
124    /// where `b` denotes the number of big ints stored.
125    pub fn set(&mut self, i: usize, v: B) {
126        let maxv: S = S::max_value();
127        match cast(v) {
128            Some(v) if v < maxv => self.smallints[i] = v,
129            _ => {
130                self.smallints[i] = maxv;
131                self.bigints.insert(i, v);
132            }
133        }
134    }
135
136    /// Iterate over sequence. Values will be returned in the big integer type (`B`).
137    pub fn iter(&self) -> Iter<'_, S, B> {
138        Iter {
139            smallints: self,
140            items: self.smallints.iter().enumerate(),
141        }
142    }
143
144    /// Decompress into a normal vector of big integers (type `B`).
145    pub fn decompress(&self) -> Vec<B> {
146        self.iter().collect()
147    }
148
149    /// Length of the sequence.
150    pub fn len(&self) -> usize {
151        self.smallints.len()
152    }
153
154    /// is the sequence empty?
155    pub fn is_empty(&self) -> bool {
156        self.smallints.is_empty()
157    }
158
159    fn real_value(&self, i: usize, v: S) -> Option<B> {
160        if v < S::max_value() {
161            cast(v)
162        } else {
163            self.bigints.get(&i).cloned()
164        }
165    }
166}

168/// Iterator over the elements of a `SmallInts` sequence.
169#[derive(Clone, Debug)]
170pub struct Iter<'a, S, B>
171where
172    S: Integer + Bounded + NumCast + Copy,
173    B: Integer + NumCast + Copy,
174    <S as Num>::FromStrRadixErr: 'a,
175    <B as Num>::FromStrRadixErr: 'a,
176{
177    smallints: &'a SmallInts<S, B>,
178    items: Enumerate<slice::Iter<'a, S>>,
179}
180
181impl<'a, S, B> Iterator for Iter<'a, S, B>
182where
183    S: 'a + Integer + Bounded + NumCast + Copy,
184    B: 'a + Integer + NumCast + Copy,
185    <S as Num>::FromStrRadixErr: 'a,
186    <B as Num>::FromStrRadixErr: 'a,
187{
188    type Item = B;
189
190    fn next(&mut self) -> Option<B> {
191        match self.items.next() {
192            Some((i, &v)) => self.smallints.real_value(i, v),
193            None => None,
194        }
195    }
196}
