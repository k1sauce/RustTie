6//! The [Full-text index in Minute space index (FM-index)](https://doi.org/10.1109/SFCS.2000.892127) and
7//! the FMD-Index for finding suffix array intervals matching a given pattern in linear time.
8//!
9//! # Examples
10//!
11//! ## Generate
12//!
13//! ```
14//! use bio::alphabets::dna;
15//! use bio::data_structures::bwt::{bwt, less, Occ};
16//! use bio::data_structures::fmindex::{FMIndex, FMIndexable};
17//! use bio::data_structures::suffix_array::suffix_array;
18//!
19//! let text = b"GCCTTAACATTATTACGCCTA$";
20//! let alphabet = dna::n_alphabet();
21//! let sa = suffix_array(text);
22//! let bwt = bwt(text, &sa);
23//! let less = less(&bwt, &alphabet);
24//! let occ = Occ::new(&bwt, 3, &alphabet);
25//! let fm = FMIndex::new(&bwt, &less, &occ);
26//! ```
27//!
28//! ## Enclose in struct
29//!
30//! `FMIndex` was designed to not forcibly own the BWT and auxiliary data structures.
31//! It can take a reference (`&`), owned structs or any of the more complex pointer types.
32//!
33//! ```
34//! use bio::alphabets::dna;
35//! use bio::data_structures::bwt::{bwt, less, Less, Occ, BWT};
36//! use bio::data_structures::fmindex::{FMIndex, FMIndexable};
37//! use bio::data_structures::suffix_array::suffix_array;
38//! use bio::utils::TextSlice;
39//!
40//! pub struct Example {
41//!     fmindex: FMIndex<BWT, Less, Occ>,
42//! }
43//!
44//! impl Example {
45//!     pub fn new(text: TextSlice) -> Self {
46//!         let alphabet = dna::n_alphabet();
47//!         let sa = suffix_array(text);
48//!         let bwt = bwt(text, &sa);
49//!         let less = less(&bwt, &alphabet);
50//!         let occ = Occ::new(&bwt, 3, &alphabet);
51//!         let fm = FMIndex::new(bwt, less, occ);
52//!         Example { fmindex: fm }
53//!     }
54//! }
55//! ```

65/// A suffix array interval.
66#[derive(
67    Default, Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize,
68)]
69pub struct Interval {
70    pub lower: usize,
71    pub upper: usize,
72}
73
74impl Interval {
75    pub fn occ<SA: SuffixArray>(&self, sa: &SA) -> Vec<usize> {
76        (self.lower..self.upper)
77            .map(|pos| sa.get(pos).expect("Interval out of range of suffix array"))
78            .collect()
79    }
80}

82/// This enum represents the potential result states
83/// from a backward_search in the fm index.  The
84/// potential variants of the enum are:
85/// Complete(Interval) — the query matched completely. The interval is the
86/// range of suffix array indices matching the query string.
87/// Partial(Intarval, usize) - some suffix of the query matched, but not the whole query.
88/// The interval returned is the range of suffix array indices for the maximal
89/// matching suffix, and the `usize` is the length of the maximal matching suffix.
90/// Absent - None suffix of the pattern matched in the text.
91#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize)]
92pub enum BackwardSearchResult {
93    Complete(Interval),
94    Partial(Interval, usize),
95    Absent,
96}
97
98pub trait FMIndexable {
99    /// Get occurrence count of symbol a in BWT[..r+1].
100    fn occ(&self, r: usize, a: u8) -> usize;
101    /// Also known as
102    fn less(&self, a: u8) -> usize;
103    fn bwt(&self) -> &BWT;
104
105    /// Perform backward search, yielding `BackwardSearchResult` enum that
106    /// contains the suffix array interval denoting exact occurrences of the given pattern
107    /// of length m in the text if it exists, or the suffix array interval denoting the
108    /// exact occurrences of a maximal matching suffix of the given pattern if it does
109    /// not exist.  If none of the pattern can be matched, the `BackwardSearchResult` is
110    /// `Absent`.
111    /// Complexity: O(m).
112    ///
113    /// # Arguments
114    ///
115    /// * `pattern` - the pattern to search
116    ///
117    /// # Example
118    ///
119    /// ```
120    /// use bio::alphabets::dna;
121    /// use bio::data_structures::bwt::{bwt, less, Occ};
122    /// use bio::data_structures::fmindex::{BackwardSearchResult, FMIndex, FMIndexable};
123    /// use bio::data_structures::suffix_array::suffix_array;
124    ///
125    /// let text = b"GCCTTAACATTATTACGCCTA$";
126    /// let alphabet = dna::n_alphabet();
127    /// let sa = suffix_array(text);
128    /// let bwt = bwt(text, &sa);
129    /// let less = less(&bwt, &alphabet);
130    /// let occ = Occ::new(&bwt, 3, &alphabet);
131    /// let fm = FMIndex::new(&bwt, &less, &occ);
132    ///
133    /// let pattern = b"TTA";
134    /// let bsr = fm.backward_search(pattern.iter());
135    ///
136    /// let positions = match bsr {
137    ///     BackwardSearchResult::Complete(sai) => sai.occ(&sa),
138    ///     BackwardSearchResult::Partial(sai, _l) => sai.occ(&sa),
139    ///     BackwardSearchResult::Absent => Vec::<usize>::new(),
140    /// };
141    ///
142    /// assert_eq!(positions, [3, 12, 9]);
143    /// ```
144    fn backward_search<'b, P: Iterator<Item = &'b u8> + DoubleEndedIterator>(
145        &self,
146        pattern: P,
147    ) -> BackwardSearchResult {
148        let (mut l, mut r) = (0, self.bwt().len() - 1);
149        // to keep track of the last "valid" search interval if
150        // there is any valid suffix match.
151        let (mut pl, mut pr) = (l, r);
152
153        // the length of the suffix we have been able to match
154        // successfully
155        let mut matched_len = 0;
156        // track if we exit early or not due to an empty
157        // search interval.
158        let mut complete_match = true;
159
160        for &a in pattern.rev() {
161            let less = self.less(a);
162            pl = l;
163            pr = r;
164            l = less + if l > 0 { self.occ(l - 1, a) } else { 0 };
165            r = less + self.occ(r, a) - 1;
166
167            // The symbol was not found if we end up with an empty interval.
168            // Terminate the LF-mapping process. In this case, also mark that
169            // we do not have a complete match.
170            if l > r {
171                complete_match = false;
172                break;
173            }
174            matched_len += 1;
175        }
176
177        // if we matched at least 1 character
178        if matched_len > 0 {
179            // if we matched the full pattern length we
180            // have a complete match
181            if complete_match {
182                BackwardSearchResult::Complete(Interval {
183                    lower: l,
184                    upper: r + 1,
185                })
186            } else {
187                // if we matched less than the full pattern length, we have
188                // a partial suffix match
189                BackwardSearchResult::Partial(
190                    Interval {
191                        lower: pl,
192                        upper: pr + 1,
193                    },
194                    matched_len,
195                )
196            }
197        } else {
198            // if we matched nothing we have an absent result
199            BackwardSearchResult::Absent
200        }
201    }
202}

204/// The Fast Index in Minute space (FM-Index, Ferragina and Manzini, 2000) for finding suffix array
205/// intervals matching a given pattern.
206#[derive(
207    Default, Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize,
208)]
209pub struct FMIndex<DBWT: Borrow<BWT>, DLess: Borrow<Less>, DOcc: Borrow<Occ>> {
210    bwt: DBWT,
211    less: DLess,
212    occ: DOcc,
213}
214
215impl<DBWT: Borrow<BWT>, DLess: Borrow<Less>, DOcc: Borrow<Occ>> FMIndexable
216    for FMIndex<DBWT, DLess, DOcc>
217{
218    fn occ(&self, r: usize, a: u8) -> usize {
219        self.occ.borrow().get(self.bwt.borrow(), r, a)
220    }
221    fn less(&self, a: u8) -> usize {
222        self.less.borrow()[a as usize]
223    }
224    /// Provide a reference to the underlying BWT.
225    fn bwt(&self) -> &BWT {
226        self.bwt.borrow()
227    }
228}

235    /// * `bwt` - the BWT
236    /// * `less` - the less array of the BWT
237    /// * `occ` - the occurence array of the BWT

239        FMIndex { bwt, less, occ }
240    }

243/// A bi-interval on suffix array of the forward and reverse strand of a DNA text.
244#[derive(
245    Default, Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize,
246)]
247pub struct BiInterval {
248    lower: usize,
249    lower_rev: usize,
250    size: usize,
251    match_size: usize,
252}
253
254impl BiInterval {
255    pub fn forward(&self) -> Interval {
256        Interval {
257            upper: self.lower + self.size,
258            lower: self.lower,
259        }
260    }
261    pub fn revcomp(&self) -> Interval {
262        Interval {
263            upper: self.lower_rev + self.size,
264            lower: self.lower_rev,
265        }
266    }
267
268    fn swapped(&self) -> BiInterval {
269        BiInterval {
270            lower: self.lower_rev,
271            lower_rev: self.lower,
272            size: self.size,
273            match_size: self.match_size,
274        }
275    }
276}

292    }

294    fn less(&self, a: u8) -> usize {
295        self.fmindex.less(a)

304impl<DBWT: Borrow<BWT>, DLess: Borrow<Less>, DOcc: Borrow<Occ>> From<FMIndex<DBWT, DLess, DOcc>>
305    for FMDIndex<DBWT, DLess, DOcc>
306{
307    /// Construct a new instance of the FMD index (see Heng Li (2012) Bioinformatics).
308    /// This expects a BWT that was created from a text over the DNA alphabet with N
309    /// (`alphabets::dna::n_alphabet()`) consisting of the
310    /// concatenation with its reverse complement, separated by the sentinel symbol `$`.
311    /// I.e., let T be the original text and R be its reverse complement.
312    /// Then, the expected text is T$R$. Further, multiple concatenated texts are allowed, e.g.
313    /// T1$R1$T2$R2$T3$R3$.
314    fn from(fmindex: FMIndex<DBWT, DLess, DOcc>) -> FMDIndex<DBWT, DLess, DOcc> {
315        let mut alphabet = dna::n_alphabet();
316        alphabet.insert(b'$');
317        assert!(
318            alphabet.is_word(fmindex.bwt()),
319            "Expecting BWT over the DNA alphabet (including N) with the sentinel $."
320        );
321
322        FMDIndex { fmindex }
323    }
324}

326impl<DBWT: Borrow<BWT>, DLess: Borrow<Less>, DOcc: Borrow<Occ>> FMDIndex<DBWT, DLess, DOcc> {
327    /// Find supermaximal exact matches (of length >= l) of given pattern that overlap position i in the pattern.
328    /// Complexity O(m) with pattern of length m.
329    ///
330    /// # Example
331    ///
332    /// ```
333    /// use bio::alphabets::dna;
334    /// use bio::data_structures::bwt::{bwt, less, Occ};
335    /// use bio::data_structures::fmindex::{FMDIndex, FMIndex};
336    /// use bio::data_structures::suffix_array::suffix_array;
337    ///
338    /// let text = b"ATTC$GAAT$";
339    /// let alphabet = dna::n_alphabet();
340    /// let sa = suffix_array(text);
341    /// let bwt = bwt(text, &sa);
342    /// let less = less(&bwt, &alphabet);
343    /// let occ = Occ::new(&bwt, 3, &alphabet);
344    /// let fm = FMIndex::new(&bwt, &less, &occ);
345    /// let fmdindex = FMDIndex::from(fm);
346    ///
347    /// let pattern = b"ATT";
348    /// let intervals = fmdindex.smems(pattern, 2, 0);
349    ///
350    /// let forward_positions = intervals[0].0.forward().occ(&sa);
351    /// let revcomp_positions = intervals[0].0.revcomp().occ(&sa);
352    /// let pattern_position = intervals[0].1;
353    /// let smem_len = intervals[0].2;
354    ///
355    /// assert_eq!(forward_positions, [0]);
356    /// assert_eq!(revcomp_positions, [6]);
357    /// assert_eq!(pattern_position, 0);
358    /// assert_eq!(smem_len, 3);
359    /// ```
360    pub fn smems(&self, pattern: &[u8], i: usize, l: usize) -> Vec<(BiInterval, usize, usize)> {
361        let curr = &mut Vec::new(); // pairs (biinterval, current match length)
362        let prev = &mut Vec::new(); // """
363        let mut matches = Vec::new(); // triples (biinterval, position on pattern, smem length)
364
365        let mut match_len = 0;
366        let mut interval = self.init_interval_with(pattern[i]);
367        if interval.size != 0 {
368            match_len += 1;
369        }
370
371        for &a in &pattern[i + 1..] {
372            // forward extend interval
373            let forward_interval = self.forward_ext(&interval, a);
374
375            // if size changed, add last interval to list
376            if interval.size != forward_interval.size {
377                curr.push((interval, match_len));
378            }
379            // if new interval size is zero, stop, as no further forward extension is possible
380            if forward_interval.size == 0 {
381                break;
382            }
383            interval = forward_interval;
384            match_len += 1;
385        }
386        // add the last non-zero interval
387        curr.push((interval, match_len));
388        // reverse intervals such that longest comes first
389        curr.reverse();
390
391        swap(curr, prev);
392        let mut j = pattern.len() as isize;
393
394        for k in (-1..i as isize).rev() {
395            let a = if k == -1 { b'$' } else { pattern[k as usize] };
396            curr.clear();
397            // size of the last confirmed interval
398            let mut last_size = -1;
399
400            for (interval, match_len) in prev.iter() {
401                // backward extend interval
402                let forward_interval = self.backward_ext(interval, a);
403
404                if (forward_interval.size == 0 || k == -1) &&
405                        // interval could not be extended further
406                        // if no interval has been extended this iteration,
407                        // interval is maximal and can be added to the matches
408                        curr.is_empty() && k < j &&
409                        match_len >= &l
410                {
411                    j = k;
412                    matches.push((*interval, (k + 1) as usize, *match_len));
413                }
414                // add _interval to curr (will be further extended next iteration)
415                if forward_interval.size != 0 && forward_interval.size as isize != last_size {
416                    last_size = forward_interval.size as isize;
417                    curr.push((forward_interval, match_len + 1));
418                }
419            }
420            if curr.is_empty() {
421                break;
422            }
423            swap(curr, prev);
424        }
425
426        matches
427    }

429    /// Find all supermaximal exact matches (of length >= l) of given pattern.
430    /// Complexity O(m^2) with pattern of length m.
431    ///
432    /// # Example
433    ///
434    /// ```
435    /// use bio::alphabets::dna;
436    /// use bio::data_structures::bwt::{bwt, less, Occ};
437    /// use bio::data_structures::fmindex::{FMDIndex, FMIndex};
438    /// use bio::data_structures::suffix_array::suffix_array;
439    ///
440    /// let text = b"ATTCGGGG$CCCCGAAT$";
441    /// let alphabet = dna::n_alphabet();
442    /// let sa = suffix_array(text);
443    /// let bwt = bwt(text, &sa);
444    /// let less = less(&bwt, &alphabet);
445    /// let occ = Occ::new(&bwt, 3, &alphabet);
446    /// let fm = FMIndex::new(&bwt, &less, &occ);
447    /// let fmdindex = FMDIndex::from(fm);
448    ///
449    /// let pattern = b"ATTGGGG";
450    /// let intervals = fmdindex.all_smems(pattern, 0);
451    /// assert_eq!(intervals.len(), 2);
452    ///
453    /// let solutions = vec![[0, 14, 0, 3], [4, 9, 3, 4]];
454    /// for (i, interval) in intervals.iter().enumerate() {
455    ///     let forward_positions = interval.0.forward().occ(&sa);
456    ///     let revcomp_positions = interval.0.revcomp().occ(&sa);
457    ///     let pattern_position = interval.1;
458    ///     let smem_len = interval.2;
459    ///     assert_eq!(
460    ///         [
461    ///             forward_positions[0],
462    ///             revcomp_positions[0],
463    ///             pattern_position,
464    ///             smem_len
465    ///         ],
466    ///         solutions[i]
467    ///     );
468    /// }
469    /// ```
470    pub fn all_smems(&self, pattern: &[u8], l: usize) -> Vec<(BiInterval, usize, usize)> {
471        let mut smems = Vec::new();
472        let mut i0 = 0;
473        while i0 < pattern.len() {
474            let mut curr_smems = self.smems(pattern, i0, l);
475            let mut next_i0 = i0 + 1; // this always works since:
476                                      // if we have a SMEM overlapping i0, it is at least 1bp long.
477                                      // If we don't have a smem, then we'll reiterate from i0+1
478            for (_, p, l) in curr_smems.iter() {
479                if p + l > next_i0 {
480                    next_i0 = p + l;
481                }
482            }
483            i0 = next_i0;
484            smems.append(&mut curr_smems);
485        }
486        smems
487    }

489    /// Initialize interval with given start character.
490    pub fn init_interval_with(&self, a: u8) -> BiInterval {
491        let comp_a = dna::complement(a);
492        let lower = self.fmindex.less(a);
493
494        BiInterval {
495            lower,
496            lower_rev: self.fmindex.less(comp_a),
497            size: self.fmindex.less(a + 1) - lower,
498            match_size: 1,
499        }
500    }
501
502    /// Initialize interval for empty pattern. The interval points at the whole suffix array.
503    pub fn init_interval(&self) -> BiInterval {
504        BiInterval {
505            lower: 0,
506            lower_rev: 0,
507            size: self.fmindex.bwt.borrow().len(),
508            match_size: 0,
509        }
510    }

512    /// Backward extension of given interval with given character.
513    pub fn backward_ext(&self, interval: &BiInterval, a: u8) -> BiInterval {
514        let mut s = 0;
515        let mut o = 0;
516        let mut l = interval.lower_rev;
517        // Interval [l(c(aP)), u(c(aP))] is a subinterval of [l(c(P)), u(c(P))] for each a,
518        // starting with the lexicographically smallest ($),
519        // then c(T) = A, c(G) = C, c(C) = G, N, c(A) = T, ...
520        // Hence, we calculate lower revcomp bounds by iterating over
521        // symbols and updating from previous one.
522        for &b in b"$TGCNAtgcna" {
523            l += s;
524            o = if interval.lower == 0 {
525                0
526            } else {
527                self.fmindex.occ(interval.lower - 1, b)
528            };
529            // calculate size
530            s = self.fmindex.occ(interval.lower + interval.size - 1, b) - o;
531            if b == a {
532                break;
533            }
534        }
535        // calculate lower bound
536        let k = self.fmindex.less(a) + o;
537
538        BiInterval {
539            lower: k,
540            lower_rev: l,
541            size: s,
542            match_size: interval.match_size + 1,
543        }
544    }
545
546    pub fn forward_ext(&self, interval: &BiInterval, a: u8) -> BiInterval {
547        let comp_a = dna::complement(a);
548
549        self.backward_ext(&interval.swapped(), comp_a).swapped()
550    }

552    /// Construct a new instance of the FMD index (see Heng Li (2012) Bioinformatics)
553    /// without checking whether the text is over the DNA alphabet with N.
554    /// This expects a BWT that was created from a text over the DNA alphabet with N
555    /// (`alphabets::dna::n_alphabet()`) consisting of the
556    /// concatenation with its reverse complement, separated by the sentinel symbol `$`.
557    /// I.e., let T be the original text and R be its reverse complement.
558    /// Then, the expected text is T$R$. Further, multiple concatenated texts are allowed, e.g.
559    /// T1$R1$T2$R2$T3$R3$.
560    ///
561    /// # Safety
562    ///
563    /// It is unsafe to construct an FMD index from an FM index that is not built on the DNA alphabet.
564    pub unsafe fn from_fmindex_unchecked(
565        fmindex: FMIndex<DBWT, DLess, DOcc>,
566    ) -> FMDIndex<DBWT, DLess, DOcc> {
567        FMDIndex { fmindex }
568    }

675        let text_builder: Vec<&[u8]> = vec![orig_text, b"$", &revcomp_text[..], b"$"];

677
678        let alphabet = dna::n_alphabet();
679        let sa = suffix_array(&text);
680        let bwt = bwt(&text, &sa);
681        let less = less(&bwt, &alphabet);
682        let occ = Occ::new(&bwt, 3, &alphabet);
683
684        let fmindex = FMIndex::new(&bwt, &less, &occ);
685        let fmdindex = FMDIndex::from(fmindex);
686        {
687            let pattern = b"AA";
688            let intervals = fmdindex.smems(pattern, 0, 0);
689            let forward = intervals[0].0.forward();
690            let revcomp = intervals[0].0.revcomp();
691            let pattern_position = intervals[0].1;
692            let smem_len = intervals[0].2;
693            assert_eq!(forward.occ(&sa), [5, 16]);
694            assert_eq!(revcomp.occ(&sa), [3, 14]);
695            assert_eq!(pattern_position, 0);
696            assert_eq!(smem_len, 2);
697        }

699            let pattern = b"CTTAA";
700            let intervals = fmdindex.smems(pattern, 1, 0);
701            assert_eq!(intervals[0].0.forward().occ(&sa), [2]);
702            assert_eq!(intervals[0].0.revcomp().occ(&sa), [14]);
703            assert_eq!(intervals[0].1, 0);
704            assert_eq!(intervals[0].2, 5);
705            assert_eq!(intervals[0].0.match_size, 5);
706        }
707        {
708            let pattern = b"CTTAA";
709            let intervals = fmdindex.smems(pattern, 1, 7);
710            assert!(intervals.is_empty());
