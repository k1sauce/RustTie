6//! `ShiftAnd` algorithm for pattern matching.
7//! Patterns may contain at most 64 symbols.
8//! Complexity: O(n) with text length n.
9//!
10//! # Example
11//!
12//! ```rust
13//! use bio::pattern_matching::shift_and;
14//! let pattern = b"AAAA";
15//! let text = b"ACGGCTAGAAAAGGCTAG";
16//! let shiftand = shift_and::ShiftAnd::new(pattern);
17//! let occ = shiftand.find_all(text).next().unwrap();
18//! assert_eq!(occ, 8);
19//! ```
20
21use std::borrow::Borrow;
22use std::iter::Enumerate;
23
24/// `ShiftAnd` algorithm.
25#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
26pub struct ShiftAnd {
27    m: usize,
28    masks: [u64; 256],
29    accept: u64,
30}
31
32impl ShiftAnd {
33    /// Create new ShiftAnd instance from a given pattern.
34    pub fn new<C, P>(pattern: P) -> Self
35    where
36        P::IntoIter: ExactSizeIterator,
37        C: Borrow<u8>,
38        P: IntoIterator<Item = C>,
39    {
40        let pattern = pattern.into_iter();
41        let m = pattern.len();
42        assert!(m <= 64, "Expecting a pattern of at most 64 symbols.");
43        let (masks, accept) = masks(pattern);
44
45        ShiftAnd { m, masks, accept }
46    }
47
48    /// Find all matches of pattern in the given text. Matches are returned as an iterator
49    /// over start positions.
50    pub fn find_all<C, T>(&self, text: T) -> Matches<'_, C, T::IntoIter>
51    where
52        C: Borrow<u8>,
53        T: IntoIterator<Item = C>,
54    {
55        Matches {
56            shiftand: self,
57            active: 0,
58            text: text.into_iter().enumerate(),
59        }
60    }
61}
62
63/// Calculate ShiftAnd masks. This function is called automatically when instantiating
64/// a new ShiftAnd for a given pattern.
65pub fn masks<C, P>(pattern: P) -> ([u64; 256], u64)
66where
67    C: Borrow<u8>,
68    P: IntoIterator<Item = C>,
69{
70    let mut masks = [0; 256];
71
72    let mut bit = 1;
73    for c in pattern {
74        masks[*c.borrow() as usize] |= bit;
75        bit *= 2;
76    }
77
78    (masks, bit / 2)
79}
80
81/// Iterator over start positions of matches.
82#[derive(Clone, Debug)]
83pub struct Matches<'a, C, T>
84where
85    C: Borrow<u8>,
86    T: Iterator<Item = C>,
87{
88    shiftand: &'a ShiftAnd,
89    active: u64,
90    text: Enumerate<T>,
91}
92
93impl<'a, C, T> Iterator for Matches<'a, C, T>
94where
95    C: Borrow<u8>,
96    T: Iterator<Item = C>,
97{
98    type Item = usize;
99
100    fn next(&mut self) -> Option<usize> {
101        for (i, c) in self.text.by_ref() {
102            self.active = ((self.active << 1) | 1) & self.shiftand.masks[*c.borrow() as usize];
103            if self.active & self.shiftand.accept > 0 {
104                return Some(i + 1 - self.shiftand.m);
105            }
106        }
107
108        None
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115    use itertools::Itertools;
116
117    #[test]
118    fn test_find_all() {
119        let text = b"dhjalkjwqnnnannanaflkjdklfj";
120        let pattern = b"qnnnannan";
121        let shiftand = ShiftAnd::new(pattern);
122        assert_eq!(shiftand.find_all(text).collect_vec(), [8]);
123    }
124
125    #[test]
126    fn test_issue_416() {
127        let text_pos_0 = b"CCTTTTTTTTTTTTTTT";
128        let pattern = b"CC";
129        let shiftand = ShiftAnd::new(pattern);
130        assert_eq!(shiftand.find_all(text_pos_0).collect_vec(), [0]);
131    }
132
133    #[test]
134    fn test_multiple_finds() {
135        let text = b"CCTCCTCC";
136        let pattern = b"CC";
137        let shiftand = ShiftAnd::new(pattern);
138        assert_eq!(shiftand.find_all(text).collect_vec(), [0, 3, 6]);
139    }
