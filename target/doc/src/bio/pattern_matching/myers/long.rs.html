1//! Myers bit-parallel approximate pattern matching with unlimited pattern length.
2//!
3//! This module implements the block-based version of the Myers algorithm
4//! and offers methods for computing the semi-global alignment.
5//! The API is exactly the same as for the 'simple' algorithm in
6//! [`bio::pattern_matching::myers`](crate::pattern_matching::myers).
7//! For patterns of up to 64 symbols, the 'simple' version is to be preferred,
8//! as it is faster.
9//! Whether the 'simple' algorithm with `u128` bit vectors outperforms the block-based
10//! implementation with `u64` bit vectors may depend on the use case.
11//!
12//! Time complexity: O(n * k)

31#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
32struct Peq<T: BitVec> {
33    peq: [T; 256],
34    /// Mask with the highest bit in the pattern set to 1;
35    /// used to compute the carry (hout) in advance_block(), which indicates
36    /// the change in edit distance betwee the characters at the boundaries
37    /// of the consecutive blocks.
38    /// Equivalent to 10^(m-1) in Myers' paper, Fig. 6.
39    /// In this block-based implementation however, we don't just replace
40    /// `m` with the fixed word size `w`, but we adapt to the the actual
41    /// length of the pattern chunk covering a block (see `States::add_block()`)
42    /// to ensure correct behavior without the wildcard padding trick.
43    high_mask: T,
44}
45
46/// Myers algorithm.
47#[derive(Default, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
48pub struct Myers<T = u64>
49where
50    T: BitVec,
51{
52    peq: Vec<Peq<T>>,
53    pub(crate) m: usize,
54    pub(crate) states_store: Vec<State<T, usize>>,
55}
56
57impl<T: BitVec> Myers<T> {
58    /// Create a new instance of Myers algorithm for a given pattern.
59    #[inline]
60    pub fn new<P, C>(pattern: P) -> Self
61    where
62        C: Borrow<u8>,
63        P: IntoIterator<Item = C>,
64        P::IntoIter: ExactSizeIterator,
65    {
66        Self::new_ambig(pattern, None, None)
67    }
68
69    #[inline]
70    pub(crate) fn new_ambig<P, C>(
71        pattern: P,
72        opt_ambigs: Option<&HashMap<u8, Vec<u8>>>,
73        opt_wildcards: Option<&[u8]>,
74    ) -> Self
75    where
76        C: Borrow<u8>,
77        P: IntoIterator<Item = C>,
78        P::IntoIter: ExactSizeIterator,
79    {
80        let w = word_size::<T>();
81        let pattern = pattern.into_iter();
82        let m = pattern.len();
83        assert!(m > 0, "Pattern is empty");
84        assert!(m <= usize::MAX / 2, "Pattern too long");
85
86        // build peq
87        let mut peq = Vec::with_capacity(ceil_div(m, w));
88        for chunk in pattern.chunks(w).into_iter() {
89            let mut peq_block = [T::zero(); 256];
90            let mut chunk_len = 0;
91            for symbol in chunk {
92                let symbol = *symbol.borrow();
93                let mask = T::one() << chunk_len;
94                // equivalent
95                peq_block[symbol as usize] |= mask;
96                // ambiguities
97                if let Some(equivalents) = opt_ambigs.and_then(|ambigs| ambigs.get(&symbol)) {
98                    for &eq in equivalents {
99                        peq_block[eq as usize] |= mask;
100                    }
101                }
102                chunk_len += 1;
103            }
104            // wildcards
105            if let Some(wildcards) = opt_wildcards {
106                for &w in wildcards {
107                    peq_block[w as usize] = T::max_value();
108                }
109            }
110
111            peq.push(Peq {
112                peq: peq_block,
113                high_mask: T::one() << (chunk_len - 1),
114            });
115        }
116
117        Myers {
118            peq,
119            m,
120            states_store: Vec::new(),
121        }
122    }
123
124    #[inline]
125    fn step(&self, state: &mut States<T>, a: u8, max_dist: usize) {
126        state.step(a, &self.peq, max_dist)
127    }
128
129    #[inline]
130    fn initial_state(&self, m: usize, max_dist: usize) -> States<T> {
131        States::new(m, max_dist)
132    }
133}
134
135#[inline]
136fn advance_block<T: BitVec>(state: &mut State<T, usize>, p: &Peq<T>, a: u8, hin: i8) -> i8 {
137    let mut eq = p.peq[a as usize];
138    let xv = eq | state.mv;
139    if hin < 0 {
140        eq |= T::one();
141    }
142    let xh = ((eq & state.pv).wrapping_add(&state.pv) ^ state.pv) | eq;
143
144    let mut ph = state.mv | !(xh | state.pv);
145    let mut mh = state.pv & xh;
146
147    // let hout = if ph & p.high_mask > T::zero() {
148    //     state.dist += 1;
149    //     1
150    // } else if mh & p.high_mask > T::zero() {
151    //     state.dist -= 1;
152    //     -1
153    // } else {
154    //     0
155    // };
156
157    // apparently faster than commented code above
158    let mut hout = ((ph & p.high_mask) != T::zero()) as i8;
159    hout -= ((mh & p.high_mask) != T::zero()) as i8;
160    state.dist = state.dist.wrapping_add(hout as usize);
161
162    ph <<= 1;
163    mh <<= 1;
164
165    if hin < 0 {
166        mh |= T::one();
167    }
168    if hin > 0 {
169        ph |= T::one();
170    }
171    // not faster:
172    // mh |= T::from_u8((hin < 0) as u8).unwrap();
173    // ph |= T::from_u8((hin > 0) as u8).unwrap();
174
175    state.pv = mh | !(xv | ph);
176    state.mv = ph & xv;
177
178    hout
179}

181/// Multi-block state
182#[derive(Default, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize)]
183pub(super) struct States<T: BitVec> {
184    states: Vec<State<T, usize>>,
185    // index of the last block (max_block_i + 1 blocks are needed to cover the full pattern)
186    max_block_i: usize,
187    // length of the remaining pattern chunk in the last block
188    // (see `add_block()`)
189    last_m: usize,
190}
191
192impl<T> States<T>
193where
194    T: BitVec,
195{
196    fn new(m: usize, max_dist: usize) -> Self {
197        let w = word_size::<T>();
198        let nblock = ceil_div(m, w);
199        let mut s = States {
200            states: Vec::with_capacity(nblock),
201            max_block_i: nblock - 1,
202            last_m: m % w,
203        };
204        // y = ceil(k/w) in Myers' paper, Fig. 9 where k = max_dist
205        // (but distances cannot exceed m)
206        let min_blocks = max(1, ceil_div(min(max_dist, m), w));
207        for _ in 0..min_blocks {
208            s.add_block(0);
209        }
210        s
211    }
212
213    /// Adds a new block to the Ukkonen band
214    /// `carry` (`vin` in Myers' paper) can be -1, 0, or 1
215    #[inline]
216    fn add_block(&mut self, carry: i8) {
217        let prev_dist = self.states.last().map(|s| s.dist).unwrap_or(0);
218        // delta: number of bits in pv/mv covered by the pattern
219        let delta = if self.states.len() == self.max_block_i && self.last_m > 0 {
220            // For the last ('bottom') block, we add the number of used bits in the last
221            // block = m % w (stored in self.last_m) and ignore the remaining bits.
222            // This strategy differs from the solution by Myers (p. 407, note 4 / p. 410, Fig. 9),
223            // where the computation starts with distance score = w * b [where b = N blocks],
224            // and both the pattern and sequence need to be padded with wild-card symbols.
225            self.last_m
226        } else {
227            word_size::<T>()
228        };
229        self.states.push(State::init(
230            prev_dist
231                .wrapping_add(delta)
232                .wrapping_add(carry as usize)
233                .to_usize()
234                .unwrap(),
235        ));
236    }
237
238    #[inline]
239    fn step(&mut self, a: u8, peq: &[Peq<T>], max_dist: usize) {
240        let mut carry = 0;
241        let mut y = self.states.len() - 1;
242
243        // compute blocks
244        for (state, block_peq) in self.states.iter_mut().zip(peq) {
245            carry = advance_block(state, block_peq, a, carry);
246        }
247
248        // adjust Ukkonen band if necessary
249        let w = word_size::<T>();
250        let last_dist = self.states[y].dist;
251        // note: this will fail if edit distances are > isize::MAX,
252        // but in practice such long patterns should not occur
253        if (last_dist as isize - carry as isize) as usize <= max_dist
254            && y < self.max_block_i
255            && (peq[y + 1].peq[a as usize] & T::one() == T::one() || carry < 0)
256        {
257            y += 1;
258            self.add_block(-carry);
259            // compute the new block as well
260            advance_block(&mut self.states[y], &peq[y], a, carry);
261        } else {
262            // note: max_dist should be <= usize::MAX - w; this is ensured beforehand
263            while y > 0 && self.states[y].dist >= max_dist + w {
264                y -= 1;
265            }
266            self.states.truncate(y + 1);
267        }
268    }
269
270    /// Returns the edit distance if known (i.e., all blocks were computed),
271    #[inline]
272    fn known_dist(&self) -> Option<usize> {
273        self.states.get(self.max_block_i).map(|s| s.dist)
274    }
275}

295impl<'a, T> StatesHandler<'a, T, usize> for LongStatesHandler<'a>
296where
297    T: BitVec + 'a,
298{
299    type TracebackHandler = LongTracebackHandler<'a, T>;
300    type TracebackColumn = States<T>;
301
302    #[inline]
303    fn init(&mut self, n: usize, m: usize) -> usize {
304        let w = word_size::<T>();
305        self.n_blocks = ceil_div(m.to_usize().unwrap(), w);
306        n * self.n_blocks
307    }
308
309    #[inline]
310    fn n_blocks(&self) -> usize {
311        self.n_blocks
312    }
313
314    #[inline]
315    fn set_max_state(&self, pos: usize, states: &mut [State<T, usize>]) {
316        let pos = pos * self.n_blocks;
317        for s in states.iter_mut().skip(pos).take(self.n_blocks) {
318            *s = State::init_max_dist();
319        }
320    }
321
322    #[inline]
323    fn add_state(
324        &self,
325        source: &Self::TracebackColumn,
326        pos: usize,
327        states: &mut [State<T, usize>],
328    ) {
329        let source = &source.states;
330        let pos = pos * self.n_blocks;
331        states[pos..pos + source.len()].clone_from_slice(source);
332
333        if source.len() < self.n_blocks {
334            // While following the traceback path, it can happen that the block to the
335            // left was not computed because it is outside of the band.
336            // Here we initialize the block below the last computed block with meaningful
337            // defaults that act as a "barrier" preventing the algorithm from moving
338            // left to this block.
339            states[pos + source.len()] = State {
340                dist: usize::MAX,
341                pv: T::zero(),
342                mv: T::zero(),
343            };
344            // Furthermore, in case arbitrary matches are requested with the
345            // `LazyMatches::..._at()` methods, we need to know if all blocks
346            // have been computed in the given column. For this, we set the
347            // distance of the last block to `usize::MAX`.
348            states[pos + self.n_blocks - 1].dist = usize::MAX;
349        }
350    }
351
352    #[inline]
353    fn dist_at(&self, pos: usize, states: &[State<T, usize>]) -> Option<usize> {
354        states.get(self.n_blocks * (pos + 1) - 1).and_then(|s| {
355            if s.dist == usize::MAX {
356                None
357            } else {
358                Some(s.dist)
359            }
360        })
361    }
362
363    #[inline]
364    fn init_traceback(
365        &self,
366        m: usize,
367        pos: usize,
368        states: &'a [State<T, usize>],
369    ) -> Option<Self::TracebackHandler> {
370        LongTracebackHandler::new(self.n_blocks, m, pos, states)
371    }
372}

374type RevColIter<'a, T> = iter::Rev<slice::Chunks<'a, State<T, usize>>>;
375
376pub(super) struct LongTracebackHandler<'a, T: BitVec> {
377    // reverse iterator used to set `col` and `left_col`
378    states_iter: iter::Chain<RevColIter<'a, T>, iter::Cycle<RevColIter<'a, T>>>,
379    // slice of blocks representing the current DP matrix column
380    col: &'a [State<T, usize>],
381    // slice of blocks representing the DP matrix column to the left
382    left_col: &'a [State<T, usize>],
383    // current block
384    block: State<T, usize>,
385    // current left block
386    left_block: State<T, usize>,
387    // index of `block` in `col`
388    block_idx: usize,
389    // index of `left_block` in `left_col`
390    left_block_idx: usize,
391    // mask with one active bit indicating the position of the last letter
392    max_mask: T,
393    // mask with one active bit indicating the current position (row) in the current block
394    pos_mask: T,
395    // Bit mask that "covers" the range of rows *below* the left cell
396    // in the current block. This is used used to initialize blocks
397    // (adjust the distance score) with `State::adjust_up_by(left_adj_mask)`.
398    left_adj_mask: T,
399    _a: PhantomData<&'a ()>,
400}
401
402impl<'a, T: BitVec> LongTracebackHandler<'a, T> {
403    #[inline]
404    fn new(n_blocks: usize, m: usize, pos: usize, states: &'a [State<T, usize>]) -> Option<Self> {
405        // length of the pattern chunk in the lowermost block
406        let mut last_m = m.to_usize().unwrap() % word_size::<T>();
407        if last_m == 0 {
408            last_m = word_size::<T>();
409        }
410        // mask with single bit indicating index of last letter in the block
411        let mask0 = T::one() << (last_m - 1);
412
413        // reverse iterator over DP matrix columns
414        // chain + cycle is needed in `FullMatches` (see comment in simple.rs)
415        let pos = n_blocks * (pos + 1);
416        let mut states_iter = states[..pos]
417            .chunks(n_blocks)
418            .rev()
419            .chain(states.chunks(n_blocks).rev().cycle());
420
421        let col = states_iter.next().unwrap();
422        let left_col = states_iter.next().unwrap();
423
424        // If the last block was not computed, we cannot obtain an alignment
425        if col.last().unwrap().dist == usize::MAX {
426            return None;
427        }
428
429        // initial data for left block
430        let (left_block_idx, left_adj_mask, max_mask) = if last_m == 1 && n_blocks > 1 {
431            (n_blocks - 2, T::zero(), T::one() << (word_size::<T>() - 1))
432        } else {
433            (n_blocks - 1, mask0, mask0)
434        };
435
436        // the left cell has to be  in diagonal position (move one up)
437        let mut left_block = left_col[left_block_idx];
438        left_block.adjust_up_by(left_adj_mask);
439
440        Some(LongTracebackHandler {
441            block: col[n_blocks - 1],
442            left_block,
443            col,
444            left_col,
445            block_idx: n_blocks - 1,
446            left_block_idx,
447            states_iter,
448            pos_mask: mask0,
449            left_adj_mask,
450            max_mask,
451            _a: PhantomData,
452        })
453    }
454
455    /// Adjust 'left_adj_mask' to cover one more cell above.
456    /// This may involve switching to the block on top
457    /// *before* the mask would cover the whole block.
458    /// Actually, only the masks and block index are adjusted,
459    /// not the `State` itself.
460    #[inline]
461    fn adjust_left_up(&mut self) -> bool {
462        let at_boundary = self.left_adj_mask & T::from_usize(0b10).unwrap() != T::zero()
463            && self.left_block_idx > 0;
464        if !at_boundary {
465            self.left_adj_mask = (self.left_adj_mask >> 1) | self.max_mask;
466        } else {
467            self.max_mask = T::one() << (word_size::<T>() - 1);
468            self.left_adj_mask = T::zero();
469            self.left_block_idx -= 1;
470        }
471        // println!("left {:064b}\nmax  {:064b}", self.left_adj_mask, self.left_max_mask);
472        at_boundary
473    }
474}

476impl<'a, T: BitVec + 'a> TracebackHandler<'a, T, usize> for LongTracebackHandler<'a, T> {
477    #[inline]
478    fn dist(&self) -> usize {
479        self.block.dist
480    }
481
482    #[inline]
483    fn left_dist(&self) -> usize {
484        self.left_block.dist
485    }
486
487    #[inline]
488    fn try_move_up(&mut self) -> bool {
489        if self.block.pv & self.pos_mask != T::zero() {
490            self.move_up();
491            return true;
492        }
493        false
494    }
495
496    #[inline]
497    fn move_up(&mut self) {
498        // (1) move up in current column
499        // If the block boundary has not been reached yet, we can shift to the
500        // upper position. Otherwise, we move to a new block (if there is one!)
501        if self.pos_mask != T::one() || self.block_idx == 0 {
502            self.block.adjust_one_up(self.pos_mask);
503            self.pos_mask >>= 1;
504        } else {
505            // move to upper block
506            self.pos_mask = T::one() << (word_size::<T>() - 1);
507            self.block_idx -= 1;
508            self.block = self.col[self.block_idx];
509        }
510        // (2) move up in left column
511        let at_boundary = self.adjust_left_up();
512        if !at_boundary {
513            self.left_block.adjust_one_up(self.pos_mask);
514        } else {
515            self.left_block = self.left_col[self.left_block_idx];
516        }
517    }
518
519    #[inline]
520    fn prepare_diagonal(&mut self) {
521        self.adjust_left_up();
522        if self.pos_mask != T::one() || self.block_idx == 0 {
523            // not at block boundary
524            self.pos_mask >>= 1;
525        } else {
526            // at boundary: move to upper block
527            self.pos_mask = T::one() << (word_size::<T>() - 1);
528            self.block_idx -= 1;
529        }
530    }
531
532    fn try_prepare_left(&mut self) -> bool {
533        if self.left_adj_mask != T::zero() {
534            // simple case: not at block boundary
535            if self.left_block.mv & self.pos_mask != T::zero() {
536                self.left_block.dist -= 1;
537                return true;
538            }
539        } else if let Some(b) = self.left_col.get(self.left_block_idx + 1) {
540            // more complicated: at lower block boundary, and there is another block below in the band;
541            if b.mv & T::one() == T::one() {
542                // move one block down again
543                let d = self.left_block.dist - 1;
544                self.left_block = *b;
545                self.left_block.dist = d;
546                return true;
547            }
548        }
549        false
550    }
551
552    #[inline]
553    fn finish_move_left(&mut self) {
554        self.col = self.left_col;
555        self.left_col = self.states_iter.next().unwrap();
556        self.block = replace(&mut self.left_block, self.left_col[self.left_block_idx]);
557        self.left_block.adjust_up_by(self.left_adj_mask);
558    }
559
560    #[inline]
561    fn done(&self) -> bool {
562        self.pos_mask == T::zero() && self.block_idx == 0
563    }
564
565    fn print_state(&self) {
566        eprintln!(
567            "--- TB dist ({:?} <-> {:?})",
568            self.left_block.dist, self.block.dist
569        );
570        eprintln!(
571            " lm {:0width$b}\nleft block={} {}\n  m {:0width$b}\ncurrent block={} {}\n",
572            self.left_adj_mask,
573            self.left_block_idx,
574            self.left_block,
575            self.pos_mask,
576            self.block_idx,
577            self.block,
578            width = word_size::<T>()
579        );
580    }
581}
