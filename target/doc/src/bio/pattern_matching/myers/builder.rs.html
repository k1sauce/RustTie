7/// Builds a Myers instance, allowing to specify ambiguities.
8///
9/// # Example:
10///
11/// This example shows how recognition of IUPAC ambiguities in patterns can be implemented:
12///
13/// ```
14/// # extern crate bio;
15/// use bio::pattern_matching::myers::MyersBuilder;
16///
17/// # fn main() {
18/// let ambigs = [
19///     (b'M', &b"AC"[..]),
20///     (b'R', &b"AG"[..]),
21///     (b'W', &b"AT"[..]),
22///     (b'S', &b"CG"[..]),
23///     (b'Y', &b"CT"[..]),
24///     (b'K', &b"GT"[..]),
25///     (b'V', &b"ACGMRS"[..]),
26///     (b'H', &b"ACTMWY"[..]),
27///     (b'D', &b"AGTRWK"[..]),
28///     (b'B', &b"CGTSYK"[..]),
29///     (b'N', &b"ACGTMRWSYKVHDB"[..]),
30/// ];
31///
32/// let mut builder = MyersBuilder::new();
33///
34/// for &(base, equivalents) in &ambigs {
35///     builder.ambig(base, equivalents);
36/// }
37///
38/// let text = b"GGATGNGCGCCATAG";
39/// let pattern = b"TRANCGG";
40/// //                *   * (mismatch)
41///
42/// let myers = builder.build_64(pattern);
43/// assert_eq!(myers.distance(text), 2);
44/// # }
45/// ```
46///
47/// Note that only ambiguities in the pattern are recognized. The reverse is not true; ambiguities
48/// in the search text are not matched by multiple symbols in the pattern. This would require
49/// specifying additional ambiguities (`builder.ambig(b'A', b"MRWVHDN")`, etc...).
50#[derive(Default, Clone, Eq, PartialEq, Debug, Serialize, Deserialize)]
51pub struct MyersBuilder {
52    ambigs: HashMap<u8, Vec<u8>>,
53    wildcards: Vec<u8>,
54}
55
56impl MyersBuilder {
57    pub fn new() -> MyersBuilder {
58        Self::default()
59    }

61    /// Allows to specify ambiguous symbols in the pattern, which will match all
62    /// given equivalent symbols in the text, as well as itself.
63    ///
64    /// The `ambig()` method can be called multiple times with the same symbol
65    /// and (potentially) different equivalents, which are simply added to the
66    /// existing list.
67    ///
68    /// # Example:
69    ///
70    /// ```
71    /// # extern crate bio;
72    /// use bio::pattern_matching::myers::MyersBuilder;
73    ///
74    /// # fn main() {
75    /// let text = b"GGATGAGCGCCATAG";
76    /// let pattern = b"TGAGCGN";
77    ///
78    /// let myers = MyersBuilder::new()
79    ///     .ambig(b'N', b"ACGT")
80    ///     .build_64(pattern);
81    ///
82    /// assert_eq!(myers.distance(text), 0);
83    /// # }
84    pub fn ambig<I, B>(&mut self, byte: u8, equivalents: I) -> &mut Self
85    where
86        I: IntoIterator<Item = B>,
87        B: Borrow<u8>,
88    {
89        let eq = self.ambigs.entry(byte).or_insert_with(Vec::new);
90        eq.extend(equivalents.into_iter().map(|b| *b.borrow()));
91        self
92    }
93
94    /// Allows to specify a wildcard symbol, that upon appearance in the search text
95    /// shall be matched by any symbol of the pattern. Multiple wildcards are possible.
96    /// For the inverse, that is, wildcards in the pattern matching any symbol in search
97    /// text, use `ambig(byte, 0..255)`.
98    ///
99    /// # Example:
100    ///
101    /// ```
102    /// # extern crate bio;
103    /// use bio::pattern_matching::myers::MyersBuilder;
104    ///
105    /// # fn main() {
106    /// let text = b"GGATGAGCG*CATAG";
107    /// let pattern = b"TGAGCGT";
108    ///
109    /// let myers = MyersBuilder::new()
110    ///     .text_wildcard(b'*')
111    ///     .build_64(pattern);
112    ///
113    /// assert_eq!(myers.distance(text), 0);
114    /// # }
115    pub fn text_wildcard(&mut self, wildcard: u8) -> &mut Self {
116        self.wildcards.push(wildcard);
117        self
118    }

120    /// Creates a Myers instance given a pattern, using `u64` as bit vector type.
121    /// Pattern length is restricted to at most 64 symbols.
122    pub fn build_64<C, P>(&self, pattern: P) -> Myers<u64>
123    where
124        C: Borrow<u8>,
125        P: IntoIterator<Item = C>,
126        P::IntoIter: ExactSizeIterator,
127    {
128        self.build(pattern)
129    }

142    /// Creates a Myers instance given a pattern, using any desired type for bit vectors.
143    /// Pattern length is restricted to the size of the bit vector `T`.
144    ///
145    /// # Example:
146    ///
147    /// ```
148    /// # extern crate bio;
149    /// use bio::pattern_matching::myers::{MyersBuilder, Myers};
150    ///
151    /// # fn main() {
152    /// let myers: Myers<u32> = MyersBuilder::new()
153    ///     .text_wildcard(b'*')
154    ///     .build(b"TGAGCG*");
155    /// // ...
156    /// # }
157    pub fn build<T, C, P>(&self, pattern: P) -> Myers<T>
158    where
159        T: BitVec,
160        C: Borrow<u8>,
161        P: IntoIterator<Item = C>,
162        P::IntoIter: ExactSizeIterator,
163    {
164        Myers::new_ambig(pattern, Some(&self.ambigs), Some(&self.wildcards))
165    }
166
