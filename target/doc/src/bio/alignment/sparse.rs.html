6//! Calculate 'sparse' alignments from kmer matches. Can be much faster than
7//! Smith-Waterman for long string, when a large enough k is used.
8//! Complexity: O(n * log(n)) for a pair of strings with n k-kmer matches. This
9//! approach is useful for generating an approximate 'backbone' alignments
10//! between two long sequences, for example in long-read alignment or
11//! genome-genome alignment. The backbone alignment can be used as-is, or can serve
12//! as a guide for a banded alignment.  By tuning k so that len(query) + len(reference) < 4^k,
13//! the number of false positive kmer matches is kept small, resulting in very
14//! fast run times for long strings.
15//!
16//! # Example
17//!
18//! ```
19//! use bio::alignment::sparse::*;
20//!
21//! let s1 =   b"ACGTACGATAGGTA";
22//! let s2 = b"TTACGTACGATAGGTATT";
23//! let k = 8;
24//! let matches = find_kmer_matches(s1, s2, k);
25//! let sparse_al = lcskpp(&matches, k);
26//! let match_path: Vec<(u32,u32)> = sparse_al.path.iter().map(|i| matches[*i]).collect();
27//! assert_eq!(match_path, vec![(0,2), (1,3), (2,4), (3,5), (4,6), (5,7), (6,8)]);
28//! assert_eq!(sparse_al.score, 14);
29

38/// Result of a sparse alignment
39#[derive(Default, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Serialize, Deserialize)]
40pub struct SparseAlignmentResult {
41    /// LCSk++ path, represented as vector of indices into the input matches vector.
42    pub path: Vec<usize>,
43    // Score of the path, which is the number of bases covered by the matched kmers.
44    pub score: u32,
45    // Full DP vector, which can generally be ignored. (It may be useful for testing purposes).
46    pub dp_vector: Vec<(u32, i32)>,
47}

49/// Sparse DP routine for Longest Common Subsequence in length k substrings.  Also known of LCSk++
50/// From LCSk++: Practical similarity metric for long strings. Filip Pavetić, Goran Žužić, Mile Šikić
51/// Paper here :https://arxiv.org/abs/1407.2407.  Original implementation here:
52/// https://github.com/fpavetic/lcskpp
53///
54/// # Arguments
55///
56/// * `matches` - a vector of tuples indicating the (string1 position, string2 position) kmer
57///   matches between the strings
58/// * `k` - the kmer length used for matching
59///
60/// # Return value
61///
62/// The method returns a `SparseAlignmentResult` struct with the following fields:
63/// * `path` is the LCSk++ path, represented as vector of indices into the input matches vector.
64/// * `score` is the score of the path, which is the number of bases covered by the matched kmers.
65/// * `dp_vector` is the full DP vector, which can generally be ignored. (It may be useful for
66///   testing purposes).
67pub fn lcskpp(matches: &[(u32, u32)], k: usize) -> SparseAlignmentResult {
68    if matches.is_empty() {
69        return SparseAlignmentResult {
70            path: Vec::new(),
71            score: 0,
72            dp_vector: Vec::new(),
73        };
74    }
75
76    let k = k as u32;
77
78    // incoming matches must be sorted to let us find the predecessor kmers by binary search.
79    for i in 1..matches.len() {
80        assert!(
81            matches[i - 1] < matches[i],
82            "incoming matches must be sorted."
83        );
84    }
85
86    let mut events: Vec<(u32, u32, u32)> = Vec::new();
87    let mut n = 0;
88
89    for (idx, &(x, y)) in matches.iter().enumerate() {
90        events.push((x, y, (idx + matches.len()) as u32));
91        events.push((x + k, y + k, idx as u32));
92
93        n = max(n, x + k);
94        n = max(n, y + k);
95    }
96    events.sort_unstable();
97
98    let mut max_col_dp: MaxBitTree<(u32, u32)> = MaxBitTree::new(n as usize);
99    let mut dp: Vec<(u32, i32)> = Vec::with_capacity(events.len());
100    let mut best_dp = (k, 0);
101
102    dp.resize(events.len(), (0, 0));
103
104    for ev in events {
105        let p = (ev.2 % matches.len() as u32) as usize;
106        let j = ev.1;
107        let is_start = ev.2 >= (matches.len() as u32);
108
109        if is_start {
110            dp[p] = (k, -1);
111            let (best_value, best_position) = max_col_dp.get(j as usize);
112            if best_value > 0 {
113                dp[p] = (k + best_value, best_position as i32);
114                best_dp = max(best_dp, (dp[p].0, p as i32));
115            }
116        } else {
117            // See if this kmer continues a different kmer
118            if ev.0 > k && ev.1 > k {
119                if let Ok(cont_idx) = matches.binary_search(&(ev.0 - k - 1, ev.1 - k - 1)) {
120                    let prev_score = dp[cont_idx].0;
121                    let candidate = (prev_score + 1, cont_idx as i32);
122                    dp[p] = max(dp[p], candidate);
123                    best_dp = max(best_dp, (dp[p].0, p as i32));
124                }
125            }
126
127            max_col_dp.set(ev.1 as usize, (dp[p].0, p as u32));
128        }
129    }
130
131    let mut traceback = Vec::new();
132    let (best_score, mut prev_match) = best_dp;
133    while prev_match >= 0 {
134        traceback.push(prev_match as usize);
135        prev_match = dp[prev_match as usize].1;
136    }
137    traceback.reverse();
138    SparseAlignmentResult {
139        path: traceback,
140        score: best_score,
141        dp_vector: dp,
142    }
143}

145#[derive(PartialEq, Eq, Ord, PartialOrd, Default, Copy, Clone)]
146struct PrevPtr {
147    plane: u32,
148    score: u32,
149    d: u32,
150    id: usize,
151    x: u32,
152    y: u32,
153}
154
155impl PrevPtr {
156    pub fn new(score: u32, x: u32, y: u32, id: usize, gap_extend: u32) -> PrevPtr {
157        let d = x + y;
158        PrevPtr {
159            plane: score + (d * gap_extend),
160            score,
161            d,
162            id,
163            x,
164            y,
165        }
166    }
167}

169/// Sparse DP routine generalizing LCSk++ method above to penalize alignment gaps.
170/// A gap is an unknown combination of mismatch, insertion and deletions, and incurs
171/// a penalty of gap_open + d * gap_extend, where d is the distance along the diagonal of the gap.
172/// # Arguments
173///
174/// * `matches` - a vector of tuples indicating the (string1 position, string2 position) kmer
175///   matches between the strings
176/// * `k` - the kmer length used for matching
177/// * `match_score` - reward for each matched base
178/// * `gap_open` - score of opening a gap, including a mismatch gap. Must be negative.
179/// * `gap_extend` - score for extending a gap along the diagonal. Must be negative.
180///
181/// # Return value
182///
183/// The method returns a `SparseAlignmentResult` struct with the following fields:
184/// * `path` is the SDP path, represented as vector of indices into the input matches vector.
185/// * `score` is the score of the path, which is the number of bases covered by the matched kmers.
186/// * `dp_vector` is the full DP vector, which can generally be ignored. (It may be useful for
187///   testing purposes).
188pub fn sdpkpp(
189    matches: &[(u32, u32)],
190    k: usize,
191    match_score: u32,
192    gap_open: i32,
193    gap_extend: i32,
194) -> SparseAlignmentResult {
195    if matches.is_empty() {
196        return SparseAlignmentResult {
197            path: Vec::new(),
198            score: 0,
199            dp_vector: Vec::new(),
200        };
201    }
202
203    let k = k as u32;
204    assert!(
205        gap_open <= 0 && gap_extend <= 0,
206        "gap parameters cannot be positive"
207    );
208    let _gap_open = (-gap_open) as u32;
209    let _gap_extend = (-gap_extend) as u32;
210
211    // incoming matches must be sorted to let us find the predecessor kmers by binary search.
212    for i in 1..matches.len() {
213        assert!(
214            matches[i - 1] < matches[i],
215            "incoming matches must be sorted"
216        );
217    }
218    // TODO: use .is_sorted when it's stabilised https://doc.rust-lang.org/std/iter/trait.Iterator.html
219
220    let mut events: Vec<(u32, u32, u32)> = Vec::new();
221    let mut n = 0;
222
223    for (idx, &(x, y)) in matches.iter().enumerate() {
224        events.push((x, y, (idx + matches.len()) as u32));
225        events.push((x + k, y + k, idx as u32));
226
227        n = max(n, x + k);
228        n = max(n, y + k);
229    }
230    events.sort_unstable();
231
232    let mut max_col_dp: MaxBitTree<PrevPtr> = MaxBitTree::new(n as usize);
233    let mut dp: Vec<(u32, i32)> = Vec::with_capacity(events.len());
234    let mut best_dp = (k, 0);
235
236    dp.resize(events.len(), (0, 0));
237
238    for ev in events {
239        let p = (ev.2 % matches.len() as u32) as usize;
240        let j = ev.1;
241        let is_start = ev.2 >= (matches.len() as u32);
242
243        if is_start {
244            // Default case -- chain starts at this node
245            dp[p] = (k * match_score, -1);
246
247            // Find best previous chain, and extend.
248            let best_prev = max_col_dp.get(j as usize);
249            if best_prev.score > 0 {
250                let prev_x = best_prev.x;
251                let prev_y = best_prev.y;
252                let cur_x = ev.0;
253                let cur_y = ev.1;
254                let gap = max(cur_x - prev_x, cur_y - prev_y);
255                let gap_penalty = if gap > 0 {
256                    _gap_open + gap * _gap_extend
257                } else {
258                    0
259                };
260
261                let reward = k * match_score;
262                let new_score = (best_prev.score + reward).saturating_sub(gap_penalty);
263
264                dp[p] = max(dp[p], (new_score, best_prev.id as i32));
265                best_dp = max(best_dp, (dp[p].0, p as i32));
266            }
267        } else {
268            // See if this kmer continues a different kmer
269            if ev.0 > k && ev.1 > k {
270                if let Ok(cont_idx) = matches.binary_search(&(ev.0 - k - 1, ev.1 - k - 1)) {
271                    let prev_score = dp[cont_idx].0;
272                    let candidate = (prev_score + match_score, cont_idx as i32);
273                    dp[p] = max(dp[p], candidate);
274                    best_dp = max(best_dp, (dp[p].0, p as i32));
275                }
276            }
277
278            let prev_frag = PrevPtr::new(dp[p].0, ev.0, ev.1, p, _gap_extend);
279            max_col_dp.set(ev.1 as usize, prev_frag);
280        }
281    }
282
283    let mut traceback = Vec::new();
284    let (best_score, mut prev_match) = best_dp;
285    while prev_match >= 0 {
286        traceback.push(prev_match as usize);
287        prev_match = dp[prev_match as usize].1;
288    }
289    traceback.reverse();
290    SparseAlignmentResult {
291        path: traceback,
292        score: best_score,
293        dp_vector: dp,
294    }
295}

297pub fn sdpkpp_union_lcskpp_path(
298    matches: &[(u32, u32)],
299    k: usize,
300    match_score: u32,
301    gap_open: i32,
302    gap_extend: i32,
303) -> Vec<usize> {
304    if matches.is_empty() {
305        return Vec::new();
306    }
307    let lcskpp_al = lcskpp(matches, k);
308    let sdpkpp_al = sdpkpp(matches, k, match_score, gap_open, gap_extend);
309    let pre_lcskpp = lcskpp_al
310        .path
311        .binary_search(&sdpkpp_al.path[0])
312        .unwrap_or(0);
313    let post_lcskpp = match lcskpp_al.path.binary_search(sdpkpp_al.path.last().unwrap()) {
314        Ok(ind) => ind + 1,
315        Err(_) => lcskpp_al.path.len(),
316    };
317
318    let mut path_union = Vec::new();
319    for i in 0..pre_lcskpp {
320        path_union.push(lcskpp_al.path[i]);
321    }
322    for i in 0..sdpkpp_al.path.len() {
323        path_union.push(sdpkpp_al.path[i]);
324    }
325    for i in post_lcskpp..lcskpp_al.path.len() {
326        path_union.push(lcskpp_al.path[i]);
327    }
328
329    path_union
330}

332/// Find all matches of length k between two strings, using a q-gram
333/// index. For very long reference strings, it may be more efficient to use and
334/// FMD index to generate the matches. Note that this method is mainly for
335/// demonstration & testing purposes.  For aligning many query sequences
336/// against the same reference, you should reuse the QGramIndex of the reference.
337pub fn find_kmer_matches(seq1: &[u8], seq2: &[u8], k: usize) -> Vec<(u32, u32)> {
338    if seq1.len() < seq2.len() {
339        let set = hash_kmers(seq1, k);
340        find_kmer_matches_seq1_hashed(&set, seq2, k)
341    } else {
342        let set = hash_kmers(seq2, k);
343        find_kmer_matches_seq2_hashed(seq1, &set, k)
344    }
345}

347/// Creates a HashMap containing all the k-mers in the sequence. FxHasher is used
348/// as the hash function instead of the inbuilt one. A good rolling hash function
349/// should speed up the code.
350pub fn hash_kmers(seq: &[u8], k: usize) -> HashMapFx<&[u8], Vec<u32>> {
351    let slc = seq;
352    let mut set: HashMapFx<&[u8], Vec<u32>> = HashMapFx::default();
353    for i in 0..(slc.len() + 1).saturating_sub(k) {
354        set.entry(&slc[i..i + k]).or_default().push(i as u32);
355    }
356    set
357}

359// Find all matches of length k between two strings where the first string is
360// already hashed by using the function sparse::hash_kmers
361pub fn find_kmer_matches_seq1_hashed(
362    seq1_set: &HashMapFx<&[u8], Vec<u32>>,
363    seq2: &[u8],
364    k: usize,
365) -> Vec<(u32, u32)> {
366    let mut matches = Vec::new();
367
368    for i in 0..(seq2.len() + 1).saturating_sub(k) {
369        let slc = &seq2[i..i + k];
370        if let Some(matches1) = seq1_set.get(slc) {
371            for pos1 in matches1 {
372                matches.push((*pos1, i as u32));
373            }
374        }
375    }
376
377    matches.sort_unstable();
378    matches
379}

381// Find all matches of length k between two strings where the second string is
382// already hashed by using the function sparse::hash_kmers
383pub fn find_kmer_matches_seq2_hashed(
384    seq1: &[u8],
385    seq2_set: &HashMapFx<&[u8], Vec<u32>>,
386    k: usize,
387) -> Vec<(u32, u32)> {
388    let mut matches = Vec::new();
389
390    for i in 0..(seq1.len() + 1).saturating_sub(k) {
391        let slc = &seq1[i..i + k];
392
393        if let Some(matches1) = seq2_set.get(slc) {
394            for pos1 in matches1 {
395                matches.push((i as u32, *pos1));
396            }
397        }
398    }
399
400    matches.sort_unstable();
401    matches

404pub fn expand_kmer_matches(
405    seq1: &[u8],
406    seq2: &[u8],
407    k: usize,
408    sorted_matches: &[(u32, u32)],
409    allowed_mismatches: usize,
410) -> Vec<(u32, u32)> {
411    for i in 1..sorted_matches.len() {
412        assert!(
413            sorted_matches[i - 1] < sorted_matches[i],
414            "incoming matches must be sorted"
415        );
416    }
417
418    let mut last_match_along_diagonal: HashMapFx<i32, (i32, i32)> = HashMapFx::default();
419    let mut left_expanded_matches: Vec<(u32, u32)> = sorted_matches.to_owned();
420
421    for &this_match in sorted_matches.iter() {
422        let diag = (this_match.0 as i32) - (this_match.1 as i32);
423        let min_xy = min(this_match.0, this_match.1) as i32;
424        let default_last_match = (
425            this_match.0 as i32 - min_xy - 1,
426            this_match.1 as i32 - min_xy - 1,
427        );
428        let last_match = last_match_along_diagonal
429            .get(&diag)
430            .cloned()
431            .unwrap_or(default_last_match);
432
433        let mut n_mismatches = 0;
434        let mut curr_pos = (this_match.0 as i32 - 1, this_match.1 as i32 - 1);
435        loop {
436            if last_match >= curr_pos {
437                break;
438            }
439            n_mismatches += if seq1[curr_pos.0 as usize] == seq2[curr_pos.1 as usize] {
440                0
441            } else {
442                1
443            };
444            if n_mismatches > allowed_mismatches {
445                break;
446            }
447            left_expanded_matches.push((curr_pos.0 as u32, curr_pos.1 as u32));
448            curr_pos = (curr_pos.0 - 1, curr_pos.1 - 1);
449        }
450        // We need to check until 1 position after this match, when we start our search from
451        // the next kmer match along this diagonal
452        last_match_along_diagonal.insert(diag, (this_match.0 as i32, this_match.1 as i32));
453    }
454
455    left_expanded_matches.sort_unstable();
456    let mut expanded_matches = left_expanded_matches.clone();
457    left_expanded_matches.reverse();
458
459    let mut next_match_along_diagonal: HashMapFx<i32, (u32, u32)> = HashMapFx::default();
460
461    for &this_match in &left_expanded_matches {
462        let diag = (this_match.0 as i32) - (this_match.1 as i32);
463        let max_inc = (min(
464            seq1.len() as u32 - this_match.0,
465            seq2.len() as u32 - this_match.1,
466        ) as u32)
467            .saturating_sub(k as u32 - 1);
468        let next_match = next_match_along_diagonal
469            .get(&diag)
470            .cloned()
471            .unwrap_or((this_match.0 + max_inc, this_match.1 + max_inc));
472
473        let mut n_mismatches = 0;
474        let mut curr_pos = (this_match.0 + 1, this_match.1 + 1);
475        loop {
476            // println!(" This : ({},{}), Current : ({},{}), Next : ({}, {}), Miss : {}",
477            // this_match.0, this_match.1, curr_pos.0, curr_pos.1, next_match.0, next_match.1, n_mismatches);
478            if curr_pos >= next_match {
479                break;
480            }
481            n_mismatches +=
482                if seq1[curr_pos.0 as usize + k - 1] == seq2[curr_pos.1 as usize + k - 1] {
483                    0
484                } else {
485                    1
486                };
487            if n_mismatches > allowed_mismatches {
488                break;
489            }
490            expanded_matches.push(curr_pos);
491            curr_pos = (curr_pos.0 + 1, curr_pos.1 + 1);
492        }
493
494        next_match_along_diagonal.insert(diag, this_match);
495    }
496    expanded_matches.sort_unstable();
497    expanded_matches
498}

717    fn test_expanded_matches() {
718        let x = b"GGGCAAAAAA";
719        let y = b"GGGGAAAAAA";
720        let matches = super::find_kmer_matches(x, y, 6);
721        assert_eq!(matches, vec![(4, 4)]);
722
723        let expanded_matches = super::expand_kmer_matches(x, y, 6, &matches, 1);
724        assert_eq!(
725            expanded_matches,
726            (0..5).map(|x| (x, x)).collect::<Vec<(u32, u32)>>()
727        );
728
729        let x = b"TTTTTTGGGCAAAAAA";
730        let y = b"TTTTTTGGGGAAAAAA";
731        let matches = super::find_kmer_matches(x, y, 6);
732        assert_eq!(matches, vec![(0, 0), (1, 1), (2, 2), (3, 3), (10, 10)]);
733
734        let expanded_matches = super::expand_kmer_matches(x, y, 6, &matches, 1);
735        assert_eq!(
736            expanded_matches,
737            (0..11).map(|x| (x, x)).collect::<Vec<(u32, u32)>>()
738        );
739
740        let x = b"TTTTTTCCGCAAAAAA";
741        let y = b"TTTTTTGGGGAAAAAA";
742        let matches = super::find_kmer_matches(x, y, 6);
743        assert_eq!(matches, vec![(0, 0), (10, 10)]);
744
745        let expanded_matches = super::expand_kmer_matches(x, y, 6, &matches, 1);
746        assert_eq!(
747            expanded_matches,
748            vec![(0, 0), (1, 1), (8, 8), (9, 9), (10, 10)]
749        );
750
751        let x = b"TTTTTTCGGCAAAAAA";
752        let y = b"TTTTTTGGGGAAAAAA";
753        let matches = super::find_kmer_matches(x, y, 6);
754        assert_eq!(matches, vec![(0, 0), (10, 10)]);

764                (7, 7),

766                (9, 9),
