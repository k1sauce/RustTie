6//! An implementation of Hidden Markov Models in Rust.
7//!
8//! ## Examples
9//!
10//! ### Discrete Emission Distribution
11//!
12//! We construct the example from Borodovsky & Ekisheva (2006), pp. 80 (also see
13//! [these slides](http://cecas.clemson.edu/~ahoover/ece854/refs/Gonze-ViterbiAlgorithm.pdf).
14//!
15//! ```rust
16//! use approx::assert_relative_eq;
17//! use bio::stats::hmm::discrete_emission::Model as DiscreteEmissionHMM;
18//! use bio::stats::hmm::viterbi;
19//! use bio::stats::Prob;
20//! use ndarray::array;
21//!
22//! let transition = array![[0.5, 0.5], [0.4, 0.6]];
23//! let observation = array![[0.2, 0.3, 0.3, 0.2], [0.3, 0.2, 0.2, 0.3]];
24//! let initial = array![0.5, 0.5];
25//!
26//! let hmm = DiscreteEmissionHMM::with_float(&transition, &observation, &initial)
27//!     .expect("Dimensions should be consistent");
28//! let (path, log_prob) = viterbi(&hmm, &vec![2, 2, 1, 0, 1, 3, 2, 0, 0]);
29//! let prob = Prob::from(log_prob);
30//! assert_relative_eq!(4.25e-8_f64, *prob, epsilon = 1e-9_f64);
31//! ```
32//!
33//! ### Continuous (Gaussian) Emission Distribution
34//!
35//! ```rust
36//! use approx::assert_relative_eq;
37//! use bio::stats::hmm::univariate_continuous_emission::GaussianModel as GaussianHMM;
38//! use bio::stats::hmm::viterbi;
39//! use bio::stats::Prob;
40//! use ndarray::array;
41//! use statrs::distribution::Normal;
42//!
43//! let transition = array![[0.5, 0.5], [0.4, 0.6]];
44//! let observation = vec![
45//!     Normal::new(0.0, 1.0).unwrap(),
46//!     Normal::new(2.0, 1.0).unwrap(),
47//! ];
48//! let initial = array![0.5, 0.5];
49//!
50//! let hmm = GaussianHMM::with_float(&transition, observation, &initial)
51//!     .expect("Dimensions should be consistent");
52//! let (path, log_prob) = viterbi(
53//!     &hmm,
54//!     &vec![-0.1, 0.1, -0.2, 0.5, 0.8, 1.1, 1.2, 1.5, 0.5, 0.2],
55//! );
56//! let prob = Prob::from(log_prob);
57//! assert_relative_eq!(2.64e-8_f64, *prob, epsilon = 1e-9_f64);
58//! ```
59//!
60//! ### Trainning a Discrete Emission Model with Baum-Welch algorithm
61//!
62//! We construct the example from Jason Eisner lecture which can be followed along
63//! with his spreadsheet ([link](http://www.cs.jhu.edu/~jason/papers/#eisner-2002-tnlp)).
64//! Take a look at tests in source file.
65//!
66//! ## Numeric Stability
67//!
68//! The implementation uses log-scale probabilities for numeric stability.
69//!
70//! ## Limitations
71//!
72//! Currently, only discrete and single-variate Gaussian continuous HMMs are implemented.
73//! Also, only dense transition matrices and trainning of discrete models are supported.
74//!
75//! ## References
76//!
77//! - Rabiner, Lawrence R. "A tutorial on hidden Markov models and selected applications
78//!   in speech recognition." Proceedings of the IEEE 77, no. 2 (1989): 257-286.

207impl Iterator for StateTransitionIter {
208    type Item = StateTransition;
209
210    fn next(&mut self) -> Option<StateTransition> {
211        let cur_b = self.nxt_b;
212        let cur_a = self.nxt_a;
213        if self.nxt_b < self.max {
214            self.nxt_b += 1;
215            Some(StateTransition::new(State(cur_a), State(cur_b)))
216        } else if self.nxt_a < self.max {
217            self.nxt_b = 0;
218            self.nxt_a += 1;
219            Some(StateTransition::new(State(cur_a), State(cur_b)))
220        } else {
221            None
222        }
223    }

226/// A trait for Hidden Markov Models (HMM) with generic `Observation` type.
227///
228/// Rabiner (1989) defines a Hidden Markov Model λ as the tiple (*A*, *B*, π) of transition matrix
229/// *A*, emission probabilities *B*, and initial state distribution π.  This has been generalized
230/// in `Model` such that you implement `transition_prob()`, `observation_prob()`, and
231/// `initial_prob()` (and the other methods; implementation of `transition_prob_idx()` can
232/// optionally be implemented and your implementation of `transition_prob()` can then panic).
233///
234/// The inference algorithm implementations `viterbi()`, `forward()`, and `backward()` will work
235/// with any implementation.
236///
237/// Consequently, this allows for the implementation of HMMs with both discrete and continuous
238/// emission distributions.
239#[allow(unconditional_recursion)]
240pub trait Model<Observation> {
241    /// The number of states in the model.
242    fn num_states(&self) -> usize;
243
244    /// Return iterator over the states of an HMM.
245    fn states(&self) -> StateIter;
246
247    /// Returns an iterator of all transitions.
248    fn transitions(&self) -> StateTransitionIter;
249
250    /// Transition probability between two states `from` and `to`.
251    fn transition_prob(&self, from: State, to: State) -> LogProb;
252
253    /// Transition probability between two states `from` and `to` for observation with index
254    /// `_to_idx` (index of `to`).
255    ///
256    /// This feature comes in handy in several applications of HMMs to biological sequences.
257    /// One prominent one is how XHMM by Fromer et al. (2014) uses the distance between target
258    /// regions for adjusting the transition probabilities.
259    ///
260    /// The default implementation return the result of the position-independent
261    /// `transition_prob()`.
262    fn transition_prob_idx(&self, from: State, to: State, _to_idx: usize) -> LogProb {
263        self.transition_prob(from, to)
264    }
265
266    /// Initial probability given the HMM `state`.
267    fn initial_prob(&self, state: State) -> LogProb;
268
269    /// Probability for the given observation in the given state.
270    fn observation_prob(&self, state: State, observation: &Observation) -> LogProb;
271
272    /// End probability given the HMM `state`.
273    fn end_prob(&self, _state: State) -> LogProb {
274        self.end_prob(_state)
275    }
276
277    fn has_end_state(&self) -> bool {
278        false
279    }
280}

282/// Compute the probability Viterbi matrix and the pointers to the origin.
283fn viterbi_matrices<O, M: Model<O>>(
284    hmm: &M,
285    observations: &[O],
286) -> (Array2<LogProb>, Array2<usize>) {
287    // The matrix with probabilities.
288    let mut vals = Array2::<LogProb>::zeros((observations.len(), hmm.num_states()));
289    // For each cell in `vals`, a pointer to the row in the previous column (for the traceback).
290    let mut from = Array2::<usize>::zeros((observations.len(), hmm.num_states()));
291
292    // Compute matrix.
293    for (i, o) in observations.iter().enumerate() {
294        if i == 0 {
295            // Initial column.
296            for s in hmm.states() {
297                vals[[0, *s]] = hmm.initial_prob(s) + hmm.observation_prob(s, o);
298                from[[0, *s]] = *s;
299            }
300        } else {
301            // Subsequent columns.
302            for j in hmm.states() {
303                let x = vals
304                    .index_axis(Axis(0), i - 1)
305                    .iter()
306                    .enumerate()
307                    .map(|(a, p)| (State(a), p))
308                    .max_by(|(a, &x), (b, &y)| {
309                        if x.is_zero() && y.is_zero() {
310                            Ordering::Equal
311                        } else if x.is_zero() {
312                            Ordering::Less
313                        } else if y.is_zero() {
314                            Ordering::Greater
315                        } else {
316                            (x + hmm.transition_prob_idx(*a, j, i))
317                                .partial_cmp(&(y + hmm.transition_prob_idx(*b, j, i)))
318                                .unwrap()
319                        }
320                    })
321                    .map(|(x, y)| (x, *y))
322                    .unwrap();
323                vals[[i, *j]] =
324                    x.1 + hmm.transition_prob_idx(x.0, j, i) + hmm.observation_prob(j, o);
325                from[[i, *j]] = *x.0;
326            }
327        }
328    }
329
330    (vals, from)
331}

333fn viterbi_traceback(vals: Array2<LogProb>, from: Array2<usize>) -> (Vec<State>, LogProb) {
334    // Traceback through matrix.
335    let n = vals.len_of(Axis(0));
336    let mut result: Vec<State> = Vec::new();
337    let mut curr = 0;
338    let mut res_prob = LogProb::ln_zero();
339    for (i, col) in vals.axis_iter(Axis(0)).rev().enumerate() {
340        if i == 0 {
341            let tmp = col
342                .iter()
343                .enumerate()
344                .max_by_key(|&(_, item)| OrderedFloat(**item))
345                .unwrap();
346            curr = tmp.0;
347            res_prob = *tmp.1;
348        } else {
349            curr = from[[n - i, curr]];
350        }
351        result.push(State(curr));
352    }
353    result.reverse();
354
355    (result, res_prob)
356}

358/// Execute Viterbi algorithm on the given slice of `Observation` values to get the maximum a
359/// posteriori (MAP) probability.
360///
361/// ## Arguments
362///
363/// - `hmm` - the `Model` to run the Viterbi algorithm on
364/// - `observations` - a slice of observation values to use in the algorithm
365///
366/// ## Result
367///
368/// The resulting pair *(s, p)* is the `Vec<State>` of most probable states given `hmm`
369/// and `observations` as well as the probability (as `LogProb`) of path `s`.
370///
371/// ## Type Parameters
372///
373/// - `O` - the observation type
374/// - `M` - type `Model` type
375pub fn viterbi<O, M: Model<O>>(hmm: &M, observations: &[O]) -> (Vec<State>, LogProb) {
376    let (vals, from) = viterbi_matrices(hmm, observations);
377    viterbi_traceback(vals, from)
378}

380/// Execute the forward algorithm and return the forward probabilites as `LogProb` values
381/// and the resulting forward probability.
382///
383/// ## Arguments
384///
385/// - `hmm` - the `Model` to run the forward algorithm on
386/// - `observations` - a slice of observation values to use in the algorithm
387///
388/// ## Result
389///
390/// The resulting pair (*P*, *p*) is the forward probability table (`P[[s, o]]` is the entry
391/// for state `s` and observation `o`) and the overall probability for `observations` (as
392/// `LogProb`).
393///
394/// ## Type Parameters
395///
396/// - `O` - the observation type
397/// - `M` - type `Model` type
398pub fn forward<O, M: Model<O>>(hmm: &M, observations: &[O]) -> (Array2<LogProb>, LogProb) {
399    // The matrix with probabilities.
400    let mut vals = Array2::<LogProb>::zeros((observations.len(), hmm.num_states()));
401
402    // Compute matrix.
403    for (i, o) in observations.iter().enumerate() {
404        if i == 0 {
405            // Initial column.
406            for s in hmm.states() {
407                vals[[0, *s]] = hmm.initial_prob(s) + hmm.observation_prob(s, o);
408            }
409        } else {
410            // Subsequent columns.
411            for j in hmm.states() {
412                let xs = hmm
413                    .states()
414                    .map(|k| {
415                        vals[[i - 1, *k]]
416                            + hmm.transition_prob_idx(k, j, i)
417                            + hmm.observation_prob(j, o)
418                        // + maybe_initial
419                    })
420                    .collect::<Vec<LogProb>>();
421                vals[[i, *j]] = LogProb::ln_sum_exp(&xs);
422            }
423        }
424    }
425
426    // Compute final probability.
427
428    let prob_vec_final = hmm
429        .states()
430        .map(|k| vals[[observations.len() - 1, *k]] + hmm.end_prob(k))
431        .collect::<Vec<LogProb>>();
432
433    let prob = LogProb::ln_sum_exp(&prob_vec_final);
434    // let prob = LogProb::ln_sum_exp(vals.row(observations.len() - 1).to_slice().unwrap());
435
436    (vals, prob)
437}

439pub fn backward<O, M: Model<O>>(hmm: &M, observations: &[O]) -> (Array2<LogProb>, LogProb) {
440    // The matrix with probabilities.
441    let mut vals = Array2::<LogProb>::zeros((observations.len(), hmm.num_states()));
442    let mut prob_vec_final = vec![];
443
444    // Compute matrix.
445    let n = observations.len();
446    for (i, o) in observations.iter().rev().enumerate() {
447        if i == 0 {
448            for j in hmm.states() {
449                vals[[0, *j]] = hmm.end_prob(j);
450            }
451
452            for j in hmm.states() {
453                let xs = hmm
454                    .states()
455                    .map(|k| {
456                        vals[[i, *k]]
457                            + hmm.transition_prob_idx(j, k, n - i)
458                            + hmm.observation_prob(k, o)
459                    })
460                    .collect::<Vec<LogProb>>();
461                if observations.len() > 1 {
462                    vals[[i + 1, *j]] = LogProb::ln_sum_exp(&xs);
463                } else {
464                    prob_vec_final = hmm
465                        .states()
466                        .map(|k| vals[[i, *k]] + hmm.initial_prob(k) + hmm.observation_prob(k, o))
467                        .collect::<Vec<LogProb>>();
468                }
469            }
470        } else if i == (observations.len() - 1) {
471            prob_vec_final = hmm
472                .states()
473                .map(|k| vals[[i, *k]] + hmm.initial_prob(k) + hmm.observation_prob(k, o))
474                .collect::<Vec<LogProb>>();
475        } else {
476            // Previous columns.
477            for j in hmm.states() {
478                let xs = hmm
479                    .states()
480                    .map(|k| {
481                        vals[[i, *k]]
482                            + hmm.transition_prob_idx(j, k, n - i)
483                            + hmm.observation_prob(k, o)
484                    })
485                    .collect::<Vec<LogProb>>();
486                vals[[i + 1, *j]] = LogProb::ln_sum_exp(&xs);
487            }
488        }
489    }
490
491    let prob = LogProb::ln_sum_exp(&prob_vec_final);
492
493    (vals, prob)
494}

496/// Execute **one step** of Baum-Welch algorithm to find the maximum likelihood estimate of the parameters of a HMM given a set of observed
497/// feature vector and return the estimated initial state distribution (*π**), estimated transition matrix (*A**),
498///  estimated emission probabilities matrix (*B**) and end probabilities vector (if the model has declared an end state beforehand).
499/// This function doesn't update the HMM parameters in the model and has been implemented for Discrete Emissions Models only.
500/// It return values as `LogProb`.
501///
502/// ## Arguments
503///
504/// - `hmm` - the `Model` to run the baum-welch or expected maximization algorithm on. It has to be a Discrete Model with a `Trainable` trait implemented.
505/// - `observations` - a slice of observation values to use in the algorithm
506///
507/// ## Result
508///
509/// The resulting tuple (*π**, *A**, B*, E*) is the estimated initial probability table (`P[s]`),
510/// the estimated transitions probability table (`P[[s, o]]` is the entry
511/// for state `s1` and other state `s2`), the estimated emission probability table (`P[[s, s]]` is the entry
512/// for state `s` and observation class `o`) and if we specify an end probability when building the model,
513/// E* is the estimated end probabilities. Otherwise, E* is a vector with size equal to initial probability
514/// and all values set to LogProb(1.0).  The values in all outputs are shown as `LogProb`.
515///
516/// ## Type Parameters
517///
518/// - `O` - the observation type (only discrete emissions)
519/// - `M` - type `Model` type
520pub fn baum_welch<O: Debug + Eq + Hash + Ord, M: Model<O>>(
521    hmm: &M,
522    observations: &[O],
523) -> (
524    Array1<LogProb>,
525    Array2<LogProb>,
526    Array2<LogProb>,
527    Array1<LogProb>,
528) {
529    // Execute forward algorithm to calculate the alpha probabilities for each time-step.
530    // Ignore P(x)
531    let (prob_table_f, _) = forward(hmm, observations);
532    // Execute backward algorithm to calculate the beta probabilities for each time-step.
533    let (prob_table_b_cor, _) = backward(hmm, observations);
534
535    // The time-step in forward matrix goes from t=0 to t=N. Coversely, the backward matrix
536    // entries are in reverse order, i.e., from t=N to t=0.
537    // The following code puts the backward matrix in the same order as forward matrix.
538    // This is usefull when computing the alfa times beta probabilities.
539    let n = observations.len() - 1;
540    let mut prob_table_b = Array2::<LogProb>::zeros((observations.len(), hmm.num_states()));
541    for (j, el) in prob_table_b_cor.axis_iter(Axis(0)).enumerate() {
542        prob_table_b.row_mut(n - j).assign(&el);
543    }
544
545    // Product of the forward probability and backward probability which is translated
546    // into a sum in log space. Results in gamma array which is an array of size = len(Observation) x N_states representing
547    // the probability of being in state t at time j for this observations.
548    let alpha_betas = &prob_table_f + &prob_table_b;
549
550    // Log-likelihood of the observation given the model.
551    // This probability must be the same as probabilities calculated by forward or backward functions
552    let probx = LogProb::ln_sum_exp(alpha_betas.row(observations.len() - 1).to_slice().unwrap());
553
554    // Dictionary-like vec_hashs_prob_obs stores b_{i}^{*}(v_{k}) which is the expected number of times the output observations
555    // have been equal to observation v_{k} while in state i over the expected
556    // total number of times in state i.
557    let mut vec_hashs_prob_obs: Vec<BTreeMap<&O, LogProb>> = Vec::new();
558
559    let mut distinct_obs = 0;
560
561    for h in hmm.states() {
562        let mut probs_observations: BTreeMap<&O, LogProb> = BTreeMap::new();
563
564        // First, for the numerator of b_{i}^{^}(v_{k}) we sum gamma for all time steps t in which the observation o_{t}
565        // is the symbol v_{k}
566        for (t, o) in observations.iter().enumerate() {
567            // For example, in einsner example, it calculates the probabilities P(->state, observation), for example p(->C,1) or p(->C,2) or p(->C,3)
568            let p = probs_observations.entry(o).or_insert_with(LogProb::ln_zero);
569            *p = (*p).ln_add_exp(alpha_betas[[t, *h]] - probx);
570        }
571        distinct_obs = probs_observations.len();
572        vec_hashs_prob_obs.push(probs_observations);
573    }
574
575    let mut vals_xi =
576        Array2::<LogProb>::zeros((observations.len(), hmm.num_states() * hmm.num_states()));
577
578    // Calculate the arc probabilities.
579    // It's the xi equation which it's entries represent the probability of being in state i and j,
580    // in times t and t+1, respectively, given the observed sequence and parameters.
581    // The vals_xi matrix is of size (time-steps, N_states^2) to store for each observation (or time-step)
582    // the probability of being at state i and j in times t and t+1. The element vals_xi[t, k] indicates
583    // the probability in time t and k is the index encoding of the pair of states (i,j).
584    for (t, o) in observations.iter().enumerate() {
585        if t == 0 {
586            continue;
587        } else {
588            for (idxstate, j) in hmm.states().enumerate() {
589                let numerador = hmm
590                    .states()
591                    .map(|i| {
592                        prob_table_f[[t - 1, *j]]
593                            + hmm.transition_prob_idx(j, i, t)
594                            + prob_table_b[[t, *i]]
595                            + hmm.observation_prob(i, o)
596                            - probx
597                    })
598                    .collect::<Vec<LogProb>>();
599
600                vals_xi
601                    .slice_mut(s![
602                        t,
603                        idxstate * hmm.num_states()..(idxstate + 1) * hmm.num_states()
604                    ])
605                    .assign(&Array1::from(numerador));
606            }
607        }
608    }
609
610    let mut sum_p_states = Vec::new();
611    for k in hmm.states() {
612        let sum_prob_states =
613            LogProb::ln_sum_exp(&alpha_betas.column(*k).map(|x| x - probx).to_vec());
614        sum_p_states.push(sum_prob_states);
615    }
616
617    let mut observations_hat = Array2::<LogProb>::zeros((hmm.num_states(), distinct_obs));
618    let mut transitions_hat = Array2::<LogProb>::zeros((hmm.num_states(), hmm.num_states()));
619
620    for (idxstate, i) in hmm.states().enumerate() {
621        let gamma_i = LogProb::ln_sum_exp(
622            &alpha_betas
623                .column(*i)
624                .to_vec()
625                .iter()
626                .map(|x| *x - probx)
627                .collect::<Vec<LogProb>>(),
628        );
629        let end_i = if hmm.has_end_state() {
630            LogProb::ln_zero()
631        } else {
632            alpha_betas[[observations.len() - 1, *i]] - probx
633        };
634        let q = vals_xi.slice(s![
635            ..,
636            idxstate * hmm.num_states()..(idxstate + 1) * hmm.num_states()
637        ]);
638        for k in hmm.states() {
639            let sa = LogProb::ln_sum_exp(&q.column(*k).to_vec());
640            transitions_hat[[*i, *k]] = sa - gamma_i.ln_sub_exp(end_i);
641        }
642
643        let mut ind_probs = vec![];
644        for v in vec_hashs_prob_obs[*i].values() {
645            ind_probs.push(*v - gamma_i);
646        }
647        observations_hat
648            .row_mut(*i)
649            .assign(&Array1::from(ind_probs));
650    }
651
652    let pi_hat = Array1::from(
653        alpha_betas
654            .row(0)
655            .to_vec()
656            .iter()
657            .map(|x| *x - probx)
658            .collect::<Vec<LogProb>>(),
659    );
660
661    let end_hat = if hmm.has_end_state() {
662        Array1::from(
663            sum_p_states
664                .iter()
665                .zip(alpha_betas.row(observations.len() - 1).to_vec().iter())
666                .map(|(sg, g)| (*g - probx) - *sg)
667                .collect::<Vec<LogProb>>(),
668        )
669    } else {
670        Array1::from(
671            pi_hat
672                .iter()
673                .map(|_x| LogProb::from(Prob(1.0)))
674                .collect::<Vec<LogProb>>(),
675        )
676    };
677
678    (pi_hat, transitions_hat, observations_hat, end_hat)
679}

684    ///
685    /// As arguments, a set of sequences (observations) and two optional argumets: maximum number of iterations (`n_iter`) and tolerance (`tol`).
686    /// The baum-welch iterative training procedure will stop either if it reaches the tolerance of the relative log-likelihood augmentation (default `1e-6`) or

689        &self,

997            &self,
998            transition_hat: Array2<LogProb>,
999            observation_hat: Array2<LogProb>,
1000            initial_hat: Array1<LogProb>,
1001            end_hat: Array1<LogProb>,

1004            *end_prob = end_hat;

1007            *transition_prob = transition_hat;

1016        fn train_baum_welch(
1017            &self,
1018            observations: &[Vec<usize>],
1019            n_iter: Option<usize>,
1020            tol: Option<f64>,
1021        ) -> (
1022            Array1<LogProb>,
1023            Array2<LogProb>,
1024            Array2<LogProb>,
1025            Array1<LogProb>,
1026        ) {
1027            let hmm = self;
1028
1029            let tol = tol.unwrap_or(1e-6_f64);
1030            let n_iter = n_iter.unwrap_or(500);
1031
1032            let (pi_hat, transitions_hat, observations_hat, end_hat) =
1033                super::baum_welch(hmm, &observations[0]);
1034            let (
1035                mut pi_hat_ref,
1036                mut transitions_hat_ref,
1037                mut observations_hat_ref,
1038                mut end_hat_ref,
1039            ) = (pi_hat, transitions_hat, observations_hat, end_hat);
1040            let (_, mut prob_fwd_new) = super::forward(hmm, &observations[0]);
1041
1042            // initialize var llh to store the log likelihood of trainned model
1043            let mut llh: LogProb = LogProb::ln_one();
1044
1045            // Get the number of observations in first "sample"
1046            let mut obs_n = observations[0].len() as f64;
1047
1048            // Normalize the previous log likelihood computed by number of observations
1049            let mut nllh_o = LogProb::from(Prob((*prob_fwd_new / obs_n).exp()));
1050
1051            println!(
1052                "Iter num {:?} - LLH: {:?} - Normalized LLH: {:?}",
1053                0, prob_fwd_new, nllh_o
1054            );
1055
1056            for iteration in 0..(n_iter - 1) {
1057                for obs in observations.iter() {
1058                    let (pi_hat, transitions_hat, observations_hat, end_hat) = baum_welch(hmm, obs);
1059
1060                    pi_hat_ref = pi_hat.to_owned();
1061                    transitions_hat_ref = transitions_hat.to_owned();
1062                    observations_hat_ref = observations_hat.to_owned();
1063                    end_hat_ref = end_hat.to_owned();
1064
1065                    hmm.update_matrices(transitions_hat, observations_hat, pi_hat, end_hat);
1066
1067                    let (_, prob_fwd_new_ii) = super::forward(hmm, obs);
1068
1069                    llh = prob_fwd_new_ii;
1070
1071                    // Get the number of observations
1072                    obs_n = obs.len() as f64;
1073                }
1074
1075                // Normalize the new log likelihood computed in this iteration by number of observations
1076                let nllh = LogProb::from(Prob((*llh / obs_n).exp()));
1077
1078                println!(
1079                    "Iter num {:?} - LLH: {:?} - Normalized LLH: {:?}",
1080                    iteration + 1,
1081                    llh,
1082                    nllh
1083                );
1084
1085                if nllh_o >= nllh {
1086                    prob_fwd_new = llh;
1087                    // Normalize the previous log likelihood computed by number of observations
1088                    nllh_o = LogProb::from(Prob((*prob_fwd_new / obs_n).exp()));
1089                    // Skip to next iteration
1090                    continue;
1091                }
1092
1093                if nllh.ln_sub_exp(nllh_o) < LogProb::from(Prob(tol)) {
1094                    // Stop trainning if the difference between the new log like and the old is less than a threshold
1095                    break;
1096                } else {
1097                    // Otherwise, set the old log like as the new log like
1098                    prob_fwd_new = llh;
1099
1100                    // Normalize the previous log likelihood computed by number of observations
1101                    nllh_o = LogProb::from(Prob((*prob_fwd_new / obs_n).exp()));
1102                }
1103            }
1104            (
1105                pi_hat_ref,
1106                transitions_hat_ref,
1107                observations_hat_ref,
1108                end_hat_ref,
1109            )
1110        }
1111    }
