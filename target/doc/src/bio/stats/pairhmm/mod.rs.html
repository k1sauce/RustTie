1//! This module contains the implementation of a classic `PairHMM` as described in
2//! Durbin, R., Eddy, S., Krogh, A., & Mitchison, G. (1998). Biological Sequence Analysis.
3//! Current Topics in Genome Analysis 2008. http://doi.org/10.1017/CBO9780511790492.
4//! It also contains a modified variant `HomopolyPairHMM` with additional homopolymer states suited
5//! for dealing with homopolymer runs in sequencing as often encountered in Oxford Nanopore
6//! sequencing data.
7//!
8//! Traits defined in this module apply to both `PairHMM` and `HomopolyPairHMM`.
9//!
10//! # Examples
11//! ```
12//! use approx::assert_relative_eq;
13//! use bio::stats::pairhmm::{
14//!     EmissionParameters, GapParameters, PairHMM, StartEndGapParameters, XYEmission,
15//! };
16//! use bio::stats::{LogProb, Prob};
17//! use num_traits::Zero;
18//!
19//! // Two sequences for which we'd like to know if they are likely related.
20//! let x = b"AAAA";
21//! let y = b"AAAT";
22//!
23//! // For this example, we disallow gaps, so all probabilities are zero here.
24//! struct GapParams;
25//! impl GapParameters for GapParams {
26//!     fn prob_gap_x(&self) -> LogProb {
27//!         LogProb::zero()
28//!     }
29//!     fn prob_gap_y(&self) -> LogProb {
30//!         LogProb::zero()
31//!     }
32//!     fn prob_gap_x_extend(&self) -> LogProb {
33//!         LogProb::zero()
34//!     }
35//!     fn prob_gap_y_extend(&self) -> LogProb {
36//!         LogProb::zero()
37//!     }
38//! }
39//! let gap_params = GapParams;
40//!
41//! // The PairHMM instance stores the gap params, since these are constant.
42//! let mut pairhmm = PairHMM::new(&gap_params);
43//!
44//! // However, emission parameters depend on the actual sequences
45//! struct EmissionParams {
46//!     x: &'static [u8],
47//!     y: &'static [u8],
48//! }
49//!
50//! const PROB_SUBSTITUTION: f64 = 0.1;
51//! const PROB_NO_SUBSTITUION: f64 = 1. - PROB_SUBSTITUTION;
52//! impl EmissionParameters for EmissionParams {
53//!     fn prob_emit_xy(&self, i: usize, j: usize) -> XYEmission {
54//!         if self.x[i] == self.y[j] {
55//!             // if two bases match, emit a Match!
56//!             XYEmission::Match(LogProb::from(Prob(PROB_NO_SUBSTITUION)))
57//!         } else {
58//!             // otherwise emit a Mismatch!
59//!             // Note that the probability here is `mismatch / 3`, since probabilities should sum
60//!             // to 1 and there are 3 possible mismatch configurations
61//!             XYEmission::Mismatch(LogProb::from(Prob(PROB_SUBSTITUTION / 3.)))
62//!         }
63//!     }
64//!
65//!     // In this example, emitting x[i] is as likely as not observing a mismatch.
66//!     // In more complex cases, this might e.g. depend on base qualities reported by the sequencer
67//!     fn prob_emit_x(&self, i: usize) -> LogProb {
68//!         LogProb::from(Prob(PROB_NO_SUBSTITUION))
69//!     }
70//!     fn prob_emit_y(&self, j: usize) -> LogProb {
71//!         LogProb::from(Prob(PROB_NO_SUBSTITUION))
72//!     }
73//!
74//!     fn len_x(&self) -> usize {
75//!         self.x.len()
76//!     }
77//!     fn len_y(&self) -> usize {
78//!         self.y.len()
79//!     }
80//! }
81//!
82//! // Since we want to do global alignment here, disallow free start and end gaps in x.
83//! struct GlobalAlignmentMode;
84//! impl StartEndGapParameters for GlobalAlignmentMode {
85//!     fn free_start_gap_x(&self) -> bool {
86//!         false
87//!     }
88//!     fn free_end_gap_x(&self) -> bool {
89//!         false
90//!     }
91//! }
92//!
93//! // Finally calculate the probability of relatedness between x and y!
94//! let prob_related = pairhmm.prob_related(&EmissionParams { x, y }, &GlobalAlignmentMode, None);
95//!
96//! // … and compare it to a rough estimation
97//! let prob_expected = LogProb::from(Prob(PROB_NO_SUBSTITUION.powi(3) * PROB_SUBSTITUTION / 3.));
98//! assert_relative_eq!(*prob_related, *prob_expected, epsilon = 1e-5);
99//! ```

110/// Trait for parametrization of `PairHMM` emission behavior.
111pub trait EmissionParameters {
112    /// Emission probability for `(x[i], y[j])`.
113    /// Returns a tuple with probability and a boolean indicating whether emissions match
114    /// (e.g., are the same DNA alphabet letter).
115    fn prob_emit_xy(&self, i: usize, j: usize) -> XYEmission;
116
117    /// Emission probability for `(x[i], -)`.
118    fn prob_emit_x(&self, i: usize) -> LogProb;
119
120    /// Emission probability for `(-, y[j])`.
121    fn prob_emit_y(&self, j: usize) -> LogProb;
122
123    fn len_x(&self) -> usize;
124
125    fn len_y(&self) -> usize;
126}
127/// Trait needed for the `HomopolyPairHMM`, because its implementation details
128/// depend on the actual bases to distinguish between Match states.
129pub trait Emission {
130    /// Base emitted at `i` in sequence `x`.
131    /// Should be one of b'A', b'C', b'G' or b'T'.
132    fn emission_x(&self, i: usize) -> u8;
133    /// Base emitted at `i` in sequence `y`.
134    /// Should be one of b'A', b'C', b'G' or b'T'.
135    fn emission_y(&self, j: usize) -> u8;
136}

138/// Trait for parametrization of `PairHMM` gap behavior.
139pub trait GapParameters {
140    /// Probability to open gap in x.
141    fn prob_gap_x(&self) -> LogProb;
142
143    /// Probability to open gap in y.
144    fn prob_gap_y(&self) -> LogProb;
145
146    /// Probability to extend gap in x.
147    fn prob_gap_x_extend(&self) -> LogProb;
148
149    /// Probability to extend gap in y.
150    fn prob_gap_y_extend(&self) -> LogProb;
151}

153/// Trait for parametrization of `PairHMM` start and end gap behavior.
154/// This trait can be used to implement global and semiglobal alignments.
155///
156/// * global: methods return `false` and `LogProb::ln_zero()`.
157/// * semiglobal: methods return `true` and `LogProb::ln_one()`.
158pub trait StartEndGapParameters {
159    /// Probability to start at `x[i]`. This can be left unchanged if you use `free_start_gap_x` and
160    /// `free_end_gap_x`.
161    #[inline]
162    #[allow(unused_variables)]
163    fn prob_start_gap_x(&self, i: usize) -> LogProb {
164        if self.free_start_gap_x() {
165            LogProb::ln_one()
166        } else {
167            // For global alignment, this has to return 0.0.
168            LogProb::ln_zero()
169        }
170    }
171
172    /// Allow free start gap in x.
173    fn free_start_gap_x(&self) -> bool;
174
175    /// Allow free end gap in x.
176    fn free_end_gap_x(&self) -> bool;
177}

179#[derive(Copy, Clone, PartialEq, PartialOrd, Debug, Serialize, Deserialize)]
180pub enum XYEmission {
181    Match(LogProb),
182    Mismatch(LogProb),
183}
184
185impl XYEmission {
186    pub fn prob(&self) -> LogProb {
187        match *self {
188            XYEmission::Match(p) => p,
189            XYEmission::Mismatch(p) => p,
190        }
191    }
192
193    pub fn is_match(&self) -> bool {
194        match *self {
195            XYEmission::Match(_) => true,
196            XYEmission::Mismatch(_) => false,
197        }
198    }
