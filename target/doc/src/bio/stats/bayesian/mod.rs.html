18/// For each of the hypothesis tests given as posterior error probabilities
19/// (PEPs, i.e. the posterior probability of the null hypothesis), estimate the FDR
20/// for the case that all null hypotheses with at most this PEP are rejected.
21/// FDR is calculated as presented by Müller, Parmigiani, and Rice,
22/// "FDR and Bayesian Multiple Comparisons Rules" (July 2006).
23/// Johns Hopkin's University, Dept. of Biostatistics Working Papers. Working Paper 115.
24///
25/// # Returns
26///
27/// A vector of expected FDRs in the same order as the given PEPs.
28pub fn expected_fdr(peps: &[LogProb]) -> Vec<LogProb> {
29    // sort indices
30    let sorted_idx =
31        (0..peps.len()).sorted_by(|&i, &j| OrderedFloat(*peps[i]).cmp(&OrderedFloat(*peps[j])));
32    // estimate FDR
33    let mut expected_fdr = vec![LogProb::ln_zero(); peps.len()];
34    for (j, (expected_fp, i)) in LogProb::ln_cumsum_exp(sorted_idx.clone().map(|i| peps[i]))
35        .zip(sorted_idx)
36        .enumerate()
37    {
38        let fdr = LogProb(*expected_fp - ((j + 1) as f64).ln());
39        expected_fdr[i] = if fdr <= LogProb::ln_one() {
40            fdr
41        } else {
42            LogProb::ln_one()
43        };
44    }
45
46    expected_fdr
47}

49#[cfg(test)]
50mod tests {
51    use super::*;
52    use crate::stats::LogProb;
53
54    #[test]
55    fn test_expected_fdr() {
56        let peps = [
57            LogProb(0.1f64.ln()),
58            LogProb::ln_zero(),
59            LogProb(0.25f64.ln()),
60        ];
61        let fdrs = expected_fdr(&peps);
62        println!("{:?}", fdrs);
63
64        assert_relative_eq!(*fdrs[1], *LogProb::ln_zero());
65        assert_relative_eq!(*fdrs[0], *LogProb(0.05f64.ln()));
66        assert_relative_eq!(*fdrs[2], *LogProb((0.35 / 3.0f64).ln()), epsilon = 0.000001);
67    }
68}
